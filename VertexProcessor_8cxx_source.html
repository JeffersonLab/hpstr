315void VertexProcessor::closeTruthLink(Track* track, const std::map<int, int>& mcp_pdg, bool have_mcp) {
316 closure_.n_tracks++;
317
318 const auto mcp_hits = track->getMcpHits();
319 closure_.n_stored_entries += mcp_hits.size();
320
321 if (mcp_hits.empty()) {
322 closure_.n_trk_no_entry++;
323 return;
324 }
325
332 std::map<int, int> per_strip;
333 std::map<int, int> per_layer;
334 std::set<std::pair<int, int>> seen;
335 for (const auto& [layer_id, particle_id] : mcp_hits) {
336 per_strip[particle_id]++;
337 if (seen.insert(std::make_pair(layer_id, particle_id)).second) per_layer[particle_id]++;
338 }
339
340 auto plurality = [](const std::map<int, int>& counts) {
341 int best_id{-1}, best_n{0};
342 for (const auto& [particle_id, count] : counts) {
343 if (count > best_n) {
344 best_id = particle_id;
345 best_n = count;
346 }
347 }
348 return best_id;
349 };
350
351 const int strip_id = plurality(per_strip);
352 if (strip_id != plurality(per_layer)) closure_.n_trk_plurality_differs++;
353
354 if (not have_mcp) return;
355
362 auto found = mcp_pdg.find(strip_id);
363 if (found == mcp_pdg.end()) {
364 closure_.n_trk_dangling++;
365 } else if (std::abs(found->second) != 11) {
366 closure_.n_trk_not_electron++;
367 } else {
368 closure_.n_trk_electron++;
369 }
370}

372void VertexProcessor::reportTruthLinkClosure() const {
373 const auto& c = closure_;
374 auto frac = [](long num, long den) {
375 std::ostringstream os;
376 if (den > 0) os << " (" << std::fixed << std::setprecision(4) << double(num) / double(den) << ")";
377 return os.str();
378 };
379
380 std::cout << "\n===== VertexProcessor truth-link closure [" << vtxCollLcio_ << "] =====\n"
381 << " events : " << c.n_events << "\n"
382 << " with " << mcPartRelLcio_ << " : " << c.n_events_with_rel << "\n"
383 << " with " << mcPartCollLcio_ << " : " << c.n_events_with_mcp << "\n";
384
385 if (c.n_events_with_rel == 0) {
386 std::cout << " no relation collection seen -- nothing to close.\n"
387 << "==========================================================\n" << std::endl;
388 return;
389 }
390
391 std::cout << " -- relation table, whole event (" << rawHitCollLcio_ << ") --\n"
392 << " raw hits : " << c.n_raw_event << "\n"
393 << " with >=1 relation : " << c.n_raw_event_with_rel
394 << frac(c.n_raw_event_with_rel, c.n_raw_event) << "\n"
395 << " -- relation table, on track --\n"
396 << " tracks : " << c.n_tracks << "\n"
397 << " tracker hits : " << c.n_trackerhits << "\n"
398 << " raw hits : " << c.n_raw << "\n"
399 << " with >=1 relation : " << c.n_raw_with_rel
400 << frac(c.n_raw_with_rel, c.n_raw) << "\n"
401 << " with 0 relations : " << (c.n_raw - c.n_raw_with_rel)
402 << frac(c.n_raw - c.n_raw_with_rel, c.n_raw) << "\n"
403 << " SimTrackerHits reached : " << c.n_simhits << "\n"
404 << " null MCParticle : " << c.n_simhits_null_mcp
405 << frac(c.n_simhits_null_mcp, c.n_simhits) << "\n"
406 << " MCParticle id NOT in " << mcPartCollLcio_ << " : " << c.n_simhits_dangling
407 << frac(c.n_simhits_dangling, c.n_simhits) << "\n"
408 << " -- closure: LCIO walk vs stored Track::getMcpHits() --\n"
409 << " entries from the walk : " << c.n_expected_entries << "\n"
410 << " entries on the stored Track : " << c.n_stored_entries << "\n"
411 << " "
412 << (c.n_expected_entries == c.n_stored_entries ? "CLOSED" : "*** MISMATCH: entries were lost ***")
413 << "\n"
414 << " -- what the plurality resolves to (what PreselectAndCategorize2021 sees) --\n"
415 << " tracks with no mcp-hit entry : " << c.n_trk_no_entry
416 << frac(c.n_trk_no_entry, c.n_tracks) << "\n"
417 << " plurality id not in MCParticle coll : " << c.n_trk_dangling
418 << frac(c.n_trk_dangling, c.n_tracks) << "\n"
419 << " plurality id is not an e+- : " << c.n_trk_not_electron
420 << frac(c.n_trk_not_electron, c.n_tracks) << "\n"
421 << " plurality id IS an e+- : " << c.n_trk_electron
422 << frac(c.n_trk_electron, c.n_tracks) << "\n"
423 << " (the first three are indistinguishable downstream:\n"
424 << " utils::hasTruthMatch returns false for all of them)\n"
425 << " -- weighting cross-check --\n"
426 << " strip-level and layer-unique plurality disagree : " << c.n_trk_plurality_differs
427 << frac(c.n_trk_plurality_differs, c.n_tracks) << "\n"
428 << "==========================================================\n" << std::endl;
429}
