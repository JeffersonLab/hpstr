346double ApOptimizationProcessor::computeTruthSignalShape(double z, double EAp) {
347 // EAp in GeV, mass in GeV
348 // z, ztarget in mm
349 double gamma = EAp / signal_mass_;
350 double ctau = 0.08 * pow(1e-4 / eps_, 2) * 0.1 / signal_mass_; // mm
351 return exp(-(z - ztarget_) / (gamma * ctau)) / (gamma * ctau);
352}

354double ApOptimizationProcessor::computePromptYield() {
355 // double Nbin = h_data_mass_rad_->GetBinContent(h_data_mass_rad_->FindBin(signal_mass_)); // content at signal mass
356 double Nbin = h_data_mass_rad_->Integral(h_data_mass_rad_->FindBin(lowMass_), h_data_mass_rad_->FindBin(highMass_)); // content in mass window
357 double alpha = 1 / 137.0;
358 double factors = 3 * TMath::Pi() * pow(eps_, 2) / (2 * alpha);
359
360 return radFrac_ * factors * Nbin * signal_mass_ / mass_bin_width_;
361}

363double ApOptimizationProcessor::computeDisplacedYield(TH1* h_chi_eff, double EAp) {
364 double expected_signal = 0.0;
365 for (int i = 1; i <= h_chi_eff->GetNbinsX(); i++) {
366 double z = h_chi_eff->GetBinCenter(i);
367 double Ntruth = computeTruthSignalShape(z, EAp);
368 double eff = h_chi_eff->GetBinContent(i) * f_xi_eff_->Eval(z);
369 expected_signal += eff * Ntruth;
370 }
371 expected_signal *= computePromptYield();
372
373 if (debug_) std::cout << "Expected Displaced Signal Yield: " << expected_signal << std::endl;
374 return expected_signal;
375}

388std::vector<double> ApOptimizationProcessor::findZcut(TH1* h_bkg_vtxz_cut, bool fromTailFit, int iteration) {
389 std::vector<double> zcuts;
390 double A = 0.0;
391 double z0 = 0.0;
392 double slope = 0.0;
393
394 if (fromTailFit) {
395 // fit z tail to find zcut corresponding to ztail_events
396 std::vector<double> fit_params = fitZBkgTail((TH1*)h_bkg_vtxz_cut, "bkg_tail_fit" + std::to_string(iteration));
397 A = fit_params[0];
398 z0 = fit_params[1];
399 slope = fit_params[2];
400 }
401
402 for (double ztail_events = min_ztail_events_; ztail_events <= 20; ztail_events += 0.5) {
403 if (fromTailFit) {
404 // fit z tail to find zcut corresponding to ztail_events
405 double zcut = z0 - (log(ztail_events) - log(A)) / slope;
406 zcuts.push_back(zcut);
407 } else {
408 double cumulative_bkg = 0.0;
409 for (int bin = h_bkg_vtxz_cut->GetNbinsX(); bin >= min_ztail_events_; bin--) {
410 cumulative_bkg += h_bkg_vtxz_cut->GetBinContent(bin);
411 if (cumulative_bkg >= 1.0) {
412 double zcut = h_bkg_vtxz_cut->GetBinLowEdge(bin);
413 zcuts.push_back(zcut);
414 break;
415 }
416 }
417 }
418 if (!scan_zcut_) {
419 break; // only need one zcut if not scanning
420 }
421 }
422
423 return zcuts;
424}

426// Get mass resolution in GeV
427double ApOptimizationProcessor::getMassResolution(double mass) {
428 mass *= 1000.; // convert to MeV
429 if (hit_category_ == "l1l1") {
430 return (0.4840 + 0.0421 * mass) / 1000.;
431 } else if (hit_category_ == "l1l2") {
432 return (0.8124 + 0.0422 * mass) / 1000.;
433 } else {
434 return 0.005; // default 5 MeV
435 }

552double* ApOptimizationProcessor::getQuantileArray(std::pair<double, double> range, int n_quantiles,
553 double initial_cut_frac) {
554 double* quantiles = new double[n_quantiles];
555 double start_frac = 1.0;
556
557 if (debug_)
558 std::cout << "ApOptimizationProcessor::getQuantileArray: Initial cut fraction = " << initial_cut_frac
559 << std::endl;
560 if (initial_cut_frac > -999.) {
561 // recenter quantiles around initial cut fraction
562 start_frac = initial_cut_frac - (n_quantiles / 2) * step_size_;
563 if (debug_)
564 std::cout << "ApOptimizationProcessor::getQuantileArray: Start at fraction = " << start_frac << std::endl;
565 for (int i = 0; i < n_quantiles; i++) {
566 if (start_frac + step_size_ * i < 0.)
567 quantiles[i] = 0.;
568 else if (start_frac + step_size_ * i > 1.)
569 quantiles[i] = 1.;
570 else
571 quantiles[i] = start_frac + step_size_ * i;
572 }
573 if (range.first < -999.) {
574 for (int i = 0; i < n_quantiles; i++) {
575 quantiles[i] = 1 - quantiles[i];
576 }
577 }
578 } else {
579 if (range.first < -999.) {
580 for (int i = 0; i < n_quantiles; i++) {
581 quantiles[i] = 1 - step_size_ * i;
582 }
583 } else if (range.second > 999.) {
584 for (int i = 0; i < n_quantiles; i++) {
585 quantiles[i] = step_size_ * i;
586 }
587 }
588 }
589 return quantiles;
590}

592std::vector<std::pair<double, double>> ApOptimizationProcessor::getZoffsetAlpha(TH2F* h_y0_vs_z, int n_quantiles,
593 int nbins) {
594 double* quantiles = getQuantileArray(std::make_pair(0, 9999.9), n_quantiles);
595 std::vector<std::pair<double, double>> zoffset_alpha;
596 processorHistos_->addHisto2d(h_y0_vs_z);
597 for (int q = 0; q < n_quantiles; q++) {
598 TGraphErrors* g_y0_vs_z =
599 processorHistos_->configureGraph(("g_y0_vs_z_q" + std::to_string(q)).c_str(), "z_{vtx} [mm]", "y_{0} [mm]");
600 processorHistos_->addGraph(g_y0_vs_z);
601 }
602 if (debug_)
603 std::cout << "ApOptimizationProcessor::getZoffsetAlpha: Number of z bins = "
604 << int(h_y0_vs_z->GetNbinsX() / nbins) << std::endl;
605
606 for (int i = 0; i < int(h_y0_vs_z->GetNbinsX() / nbins); i++) {
607 auto h_proj = h_y0_vs_z->ProjectionY(("h_y0_vs_z_py_" + std::to_string(i)).c_str(), nbins * i + 1, nbins * (i + 1));
608
609 if (h_proj->GetEntries() < 50) {
610 if (debug_) {
611 std::cout << "ApOptimizationProcessor::getZoffsetAlpha: Skipping z slice " << i
612 << " with entries = " << h_proj->GetEntries() << std::endl;
613 }
614 continue;
615 }
616 processorHistos_->addHisto1d(h_proj);
617 double zval = ((TAxis*)h_y0_vs_z->GetXaxis())->GetBinCenter(nbins * (i + 0.5));
618
619 double* quantile_pos = new double[n_quantiles];
620 h_proj->GetQuantiles(n_quantiles, quantile_pos, quantiles);
621 for (int q = 0; q < n_quantiles; q++) {
622 processorHistos_->getGraph(("g_y0_vs_z_q" + std::to_string(q)).c_str())->AddPointError(zval, quantile_pos[q], 0.0, 0.0);
623 // double error = 0.;
624
625 // processorHistos_->getGraph(("g_y0_vs_z_q" + std::to_string(q)).c_str())->SetPointError(i, 0.0, error);
626 }
627 }
628 for (int q = 0; q < n_quantiles; q++) {
629 if (q == 0) {
630 zoffset_alpha.push_back(std::make_pair(0.0, 0.0));
631 continue;
632 }
633 TF1* f1 = new TF1(("f1_y0_vs_z_q" + std::to_string(q)).c_str(), "[0] + [1]*x", -0.5, 15);
634 processorHistos_->getGraph(("g_y0_vs_z_q" + std::to_string(q)).c_str())->Fit(f1, "QRS");
635
636 double zoffset = -f1->GetParameter(0) / f1->GetParameter(1);
637 zoffset_alpha.push_back(std::make_pair(zoffset, f1->GetParameter(1)));
638 if (debug_)
639 std::cout << "Quantile " << q << " : zoffset = " << zoffset << " alpha = " << f1->GetParameter(1)
640 << std::endl;
641 }
642 return zoffset_alpha;
643}
