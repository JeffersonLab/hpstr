76bool PreselectAndCategorize::process(IEvent*) {
77 const auto& eh{bus_.get<EventHeader>("EventHeader")};
78 int run_number = eh.getRunNumber();
79
80 event_cf_.begin_event();
81
82 const auto& vtxs{bus_.get<std::vector<Vertex*>>(vtxColl_)};
83
84 std::vector<std::tuple<Vertex,Particle,Particle>> preselected_vtx;
85 for (Vertex* vtx : vtxs) {
86 // access the indiviual Vertex, electron, and positron
87 // and add corrections to them before applying pre-selection
88 int i_ele{-1}, i_pos{-1};
89 for (int ipart = 0; ipart < vtx->getParticles().GetEntries(); ++ipart) {
90 int pdg_id = ((Particle*)vtx->getParticles().At(ipart))->getPDG();
91 if (pdg_id == 11) {
92 i_ele = ipart;
93 } else if (pdg_id == -11) {
94 i_pos = ipart;
95 }
96 }
97 if (i_ele < 0 or i_pos < 0) {
98 throw std::runtime_error("Vertex formed without either an electron or positron!");
99 }
100
101 Particle ele = *dynamic_cast<Particle*>(vtx->getParticles().At(i_ele));
102 Particle pos = *dynamic_cast<Particle*>(vtx->getParticles().At(i_pos));
103
104 // for both particles, get cluster, update, re-set cluster
105 CalCluster ele_clu = ele.getCluster();
106 CalCluster pos_clu = pos.getCluster();
107
108 ele_clu.setTime(ele_clu.getTime() - calTimeOffset_);
109 pos_clu.setTime(pos_clu.getTime() - calTimeOffset_);
110
111 ele.setCluster(&ele_clu);
112 pos.setCluster(&pos_clu);
113
114 Track ele_trk = ele.getTrack();
115 Track pos_trk = pos.getTrack();
116
117 // put tracks back into their particles
118 // with their new data
119 ele.setTrack(&ele_trk);
120 pos.setTrack(&pos_trk);
121
122 vertex_cf_.begin_event();
123 vertex_cf_.apply("ele_track_chi2ndf", ele.getTrack().getChi2Ndf() <= 20.0);
124 vertex_cf_.apply("pos_track_chi2ndf", pos.getTrack().getChi2Ndf() <= 20.0);
125 vertex_cf_.apply("vertex_chi2", vtx->getChi2() <= 20.0);
126
127 vertex_cf_.fill_nm1("ele_track_chi2ndf", ele.getTrack().getChi2Ndf());
128 vertex_cf_.fill_nm1("pos_track_chi2ndf", pos.getTrack().getChi2Ndf());
129 vertex_cf_.fill_nm1("vertex_chi2", vtx->getChi2());
130
131 if (vertex_cf_.keep()) {
132 preselected_vtx.emplace_back(*vtx, ele, pos);
133 }
134 }
135
136 n_vertices_h_->Fill(vtxs.size(), preselected_vtx.size());
137 event_cf_.apply("at_least_one_vertex", preselected_vtx.size() > 0);
138 event_cf_.apply("no_extra_vertices", preselected_vtx.size() < 2);
139 if (not event_cf_.keep()) {
140 return true;
141 }
142
143 // correct number of vertices (i.e. only one)
144 // unpack the vector of vertices into the single elements
145 auto [ vtx, ele, pos ] = preselected_vtx.at(0);
146
147 // earliest layer hit categories
148 bool eleL1{false}, eleL2{false}, posL1{false}, posL2{false};
149 Track ele_trk{ele.getTrack()}, pos_trk{pos.getTrack()};
150 _ah->InnermostLayerCheck(&ele_trk, eleL1, eleL2);
151 _ah->InnermostLayerCheck(&pos_trk, posL1, posL2);
152 bus_.set("eleL1", eleL1);
153 bus_.set("eleL2", eleL2);
154 bus_.set("posL1", posL1);
155 bus_.set("posL2", posL2);
156
157 TVector3 ele_mom(
158 ele_trk.getMomentum()[0],
159 ele_trk.getMomentum()[1],
160 ele_trk.getMomentum()[2]
161 );
162 TVector3 pos_mom(
163 pos_trk.getMomentum()[0],
164 pos_trk.getMomentum()[1],
165 pos_trk.getMomentum()[2]
166 );
167 bus_.set("psum", ele_mom.Mag() + pos_mom.Mag());
168
169 bus_.set("weight", 1.);
170 bus_.set("vertex", vtx);
171 bus_.set("ele", ele);
172 bus_.set("pos", pos);
173
174 output_tree_->Fill();
175 return true;
176}
