92void TrackBiasingTool::loadZ0Calibration(const std::string& jsonFile) {
93 std::ifstream ifs(jsonFile);
94 if (!ifs.is_open())
95 throw std::invalid_argument("TrackBiasingTool: cannot open z0 calibration file: " + jsonFile);
96 json cfg;
97 ifs >> cfg;
98
99 auto loadParams = [](const json& node, const std::string& key) -> Z0CalibParams {
100 Z0CalibParams p;
101 if (node.contains(key)) {
102 p.slope = node[key].value("slope", 0.0);
103 p.intercept = node[key].value("intercept", 0.0);
104 }
105 return p;
106 };
107
108 for (auto& [runStr, runCfg] : cfg.items()) {
109 // "default" key is treated as run 0 (always selected by largest-key-<=run logic)
110 int runKey = (runStr == "default") ? 0 : std::stoi(runStr);
111 Z0CalibSet s;
112 s.ele_top = loadParams(runCfg, "ele_top");
113 s.ele_bot = loadParams(runCfg, "ele_bot");
114 s.pos_top = loadParams(runCfg, "pos_top");
115 s.pos_bot = loadParams(runCfg, "pos_bot");
116 z0CalibByRun_[runKey] = s;
117 }
118
119 z0CalibLoaded_ = true;
120 std::cout << "TrackBiasingTool: loaded z0 calibration from " << jsonFile
121 << " (" << z0CalibByRun_.size() << " run entries)" << std::endl;
122}

152void TrackBiasingTool::updateWithCalibrateZ0(Track& trk) {
153 if (!z0CalibLoaded_) return;
154
155 bool isTop = trk.getTanLambda() > 0.0;
156 bool isEle = trk.getCharge() < 0;
157
158 const Z0CalibParams& p =
159 isEle ? (isTop ? z0CalibActive_.ele_top : z0CalibActive_.ele_bot)
160 : (isTop ? z0CalibActive_.pos_top : z0CalibActive_.pos_bot);
161
162 double tanL = trk.getTanLambda();
163 double z0 = trk.getZ0();
164 double delta = p.slope * tanL + p.intercept;
165 double z0new = z0 - delta;
166
167 if (debug_) {
168 std::cout << "TrackBiasingTool::updateWithCalibrateZ0:"
169 << " type=" << (isEle ? "ele" : "pos") << "_" << (isTop ? "top" : "bot")
170 << " |tanL|=" << tanL
171 << " slope=" << p.slope << " intercept=" << p.intercept
172 << " delta=" << delta
173 << " z0=" << z0 << " -> " << z0new << std::endl;
174 }
175
176 trk.setZ0(z0new);
177}
