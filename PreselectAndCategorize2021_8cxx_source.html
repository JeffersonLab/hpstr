162double PreselectAndCategorize2021::calculate_theta_R(const TVector3& ele_mom, const TVector3& pos_mom) const {
163 // Beam 4-momentum (assume m_e ~ 0 for the beam)
164 double theta_beam = thetaBeamMrad_ / 1000.0; // convert to radians
165 double beam_px = beamE_ * std::sin(theta_beam);
166 double beam_py = 0.0;
167 double beam_pz = beamE_ * std::cos(theta_beam);
168
169 // Inferred recoil 3-momentum from momentum conservation
170 double recoil_px = beam_px - ele_mom.X() - pos_mom.X();
171 double recoil_py = beam_py - ele_mom.Y() - pos_mom.Y();
172 double recoil_pz = beam_pz - ele_mom.Z() - pos_mom.Z();
173
174 TVector3 recoil_mom(recoil_px, recoil_py, recoil_pz);
175 double p_recoil_mag = recoil_mom.Mag();
176 if (p_recoil_mag <= 0.0) return -9999.0;
177
178 // Beam direction (unit vector)
179 TVector3 beam_dir(std::sin(theta_beam), 0.0, std::cos(theta_beam));
180
181 // Angle between recoil and beam directions, in mrad
182 double cos_theta = recoil_mom.Dot(beam_dir) / p_recoil_mag;
183 cos_theta = std::max(-1.0, std::min(1.0, cos_theta));
184 return std::acos(cos_theta) * 1000.0;
185}

187Track PreselectAndCategorize2021::createInferredTrack(Vertex* vtx, Track* ele_track, Track* pos_track) {
188
189 Track inferredTrack;
190
191 TVector3 ele_mom(ele_track->getMomentum()[0], ele_track->getMomentum()[1], ele_track->getMomentum()[2]);
192 TVector3 pos_mom(pos_track->getMomentum()[0], pos_track->getMomentum()[1], pos_track->getMomentum()[2]);
193
194 // Beam 4-momentum (assume m_e ~ 0 for the beam)
195 double theta_beam = thetaBeamMrad_ / 1000.0; // convert to radians
196 double beam_px = beamE_ * std::sin(theta_beam);
197 double beam_py = 0.0;
198 double beam_pz = beamE_ * std::cos(theta_beam);
199
200 // Inferred recoil 3-momentum from momentum conservation
201 double recoil_px = beam_px - ele_mom.X() - pos_mom.X();
202 double recoil_py = beam_py - ele_mom.Y() - pos_mom.Y();
203 double recoil_pz = beam_pz - ele_mom.Z() - pos_mom.Z();
204
205 float inferredCharge = -1; // Third track is an electron
206 float pT = std::sqrt(ele_mom.X()*ele_mom.X() + ele_mom.Z()*ele_mom.Z());
207 float cB = pT * fabs(ele_track->getOmega());
208
209
210
211 float target_pos = -1.1;
212 // Calculate track parameters
213 std::vector<float> params = TrackTools::calculateTrackParameters(
214 vtx->getX(), vtx->getY(), vtx->getZ(),
215 recoil_px, recoil_py, recoil_pz, inferredCharge, cB, target_pos);
216
217 inferredTrack.setTrackParameters(
218 params[0], // d0
219 params[1], // phi0
220 params[2], // omega
221 params[4], // tanLambda
222 params[3] // z0
223 );
224
225 TMatrixD covVertex = TrackTools::getVertexCovariance(vtx);
226
227 TMatrixD eleCovMomentum = TrackTools::momentumCovarianceFromTrack(ele_track);
228 TMatrixD posCovMomentum = TrackTools::momentumCovarianceFromTrack(pos_track);
229
230 TMatrixD covMomentum = eleCovMomentum + posCovMomentum;
231
232 // Calculate track parameter covariance
233 TMatrixD covTrackParams = TrackTools::calculateTrackCovariance(
234 vtx->getX(), vtx->getY(), vtx->getZ(),
235 recoil_px, recoil_py, recoil_pz,
236 covVertex, covMomentum,
237 inferredCharge, cB, target_pos);
238
239
240 // Order: [d0, phi0, omega, z0, tanLambda]
241 std::vector<float> covArray;
242 covArray.reserve(15);
243
244 for (int i = 0; i < 5; i++) {
245 for (int j = 0; j <= i; j++) {
246 covArray.push_back(covTrackParams(j, i));
247 }
248 }
249
250 inferredTrack.setCov(covArray);
251
252 // Set momentum
253 inferredTrack.setMomentum(recoil_px, recoil_py, recoil_pz);
254
255 return inferredTrack;

260bool PreselectAndCategorize2021::isQualityTrack(Track* trk, const Particle& pos) const {
261 // chi2/ndf <= 20
262 if (trk->getChi2Ndf() > 20.0) return false;
263
264 // n_hits
265 int nhits = trk->getTrackerHitCount();
266 if (!trk->isKalmanTrack()) nhits *= 2;
267 if (nhits < minHits_) return false;
268
269 // p >= 0.4
270 double p = trk->getP();
271 if (p < 0.4) return false;
272
273 // for electrons (charge = -1), p <= 2.9
274 if (trk->getCharge() < 0 && p > 2.9) return false;
275
276 // Timing cuts relative to preselection positron
277 double trk_time = trk->getTrackTime();
278 double pos_trk_time = pos.getTrack().getTrackTime();
279 double pos_clu_time = pos.getCluster().getTime();
280
281 if (!disableTimingCuts_) {
282 if (trk->getCharge() < 0) {
283 if (std::fabs(trk_time - pos_clu_time) > time_cuts_[0]) return false;
284 if (std::fabs(trk_time - pos_trk_time) > time_cuts_[2]) return false;
285 } else if (trk->getCharge() > 0) {
286 if (std::fabs(trk_time - pos_clu_time) > time_cuts_[1]) return false;
287 if (std::fabs(trk_time - pos_trk_time) > time_cuts_[2]) return false;
288 }
289 }
290
291 return true;
292}

294void PreselectAndCategorize2021::calculatePairwiseQuantities(
295 const std::vector<Track*>& all_tracks,
296 Particle& ele,
297 Particle& pos)
298{
299 // Select quality tracks with good timing relative to preselection positron
300 std::vector<Track> electrons, positrons;
301 std::vector<Track> all_good_tracks;
302
303 // include the preselected electron and positron first
304 Track ele_trk = ele.getTrack();
305 Track pos_trk = pos.getTrack();
306
307 if (ele_trk.getCharge() < 0) {
308 electrons.push_back(ele_trk);
309 if (saveAllTracks_) all_good_tracks.push_back(ele_trk);
310 }
311
312 if (pos_trk.getCharge() > 0) {
313 positrons.push_back(pos_trk);
314 if (saveAllTracks_) all_good_tracks.push_back(pos_trk);
315 }
316
317 // Track which tracks from the collection we've already included
318 std::set<int> included_track_ids;
319 if (ele_trk.getID() >= 0) included_track_ids.insert(ele_trk.getID());
320 if (pos_trk.getID() >= 0) included_track_ids.insert(pos_trk.getID());
321
322 // Then check additional tracks from the collection
323 for (auto* trk : all_tracks) {
324 // Skip if this is the same track as preselected electron or positron
325 if (trk->getID() >= 0 && included_track_ids.count(trk->getID()) > 0) {
326 continue;
327 }
328
329 if (!isQualityTrack(trk, pos)) continue;
330
331 // Store a copy of the good track
332 if (saveAllTracks_) {
333 all_good_tracks.push_back(*trk);
334 }
335
336 if (trk->getCharge() < 0) {
337 electrons.push_back(*trk);
338 } else if (trk->getCharge() > 0) {
339 positrons.push_back(*trk);
340 }
341 }
342
343 // Save all good tracks if enabled
344 if (saveAllTracks_) {
345 bus_.set("all_preselected_tracks", all_good_tracks);
346 }
347
348 // Initialize extrema values
349 double min_dTanLambda = 9999.0, max_dTanLambda = -9999.0;
350 double min_opening = 9999.0, max_opening = -9999.0;
351 double min_mass = 9999.0, max_mass = -9999.0;
352 double min_asym = 9999.0, max_asym = -9999.0;
353 double min_dPhi = 9999.0, max_dPhi = -9999.0;
354
355 const double me = 0.000511;
356
357 // Loop over all e+/e- pairs
358 for (auto& ele_track : electrons) {
359 const std::vector<double>& ele_mom = ele_track.getMomentum();
360 TVector3 pe(ele_mom[0], ele_mom[1], ele_mom[2]);
361 double pe_mag = pe.Mag();
362 double ele_tanLambda = ele_track.getTanLambda();
363 double ele_phi0 = ele_track.getPhi();
364
365 for (auto& pos_track : positrons) {
366 const std::vector<double>& pos_mom = pos_track.getMomentum();
367 TVector3 pp(pos_mom[0], pos_mom[1], pos_mom[2]);
368 double pp_mag = pp.Mag();
369 double pos_tanLambda = pos_track.getTanLambda();
370 double pos_phi0 = pos_track.getPhi();
371
372 // TanLambda difference
373 double dTanLambda = std::fabs(ele_tanLambda - pos_tanLambda);
374 min_dTanLambda = std::min(min_dTanLambda, dTanLambda);
375 max_dTanLambda = std::max(max_dTanLambda, dTanLambda);
376
377 // Phi difference
378 double dPhi = std::fabs(ele_phi0 - pos_phi0);
379 min_dPhi = std::min(min_dPhi, dPhi);
380 max_dPhi = std::max(max_dPhi, dPhi);
381
382 // Opening angle
383 double cos_theta = pe.Dot(pp) / (pe_mag * pp_mag);
384 cos_theta = std::max(-1.0, std::min(1.0, cos_theta));
385 double opening_angle = std::acos(cos_theta);
386 min_opening = std::min(min_opening, opening_angle);
387 max_opening = std::max(max_opening, opening_angle);
388
389 // Invariant mass
390 double Ee = std::sqrt(pe_mag * pe_mag + me * me);
391 double Ep = std::sqrt(pp_mag * pp_mag + me * me);
392 TVector3 p_tot = pe + pp;
393 double E_tot = Ee + Ep;
394 double m2 = E_tot * E_tot - p_tot.Mag2();
395 double mass = (m2 > 0) ? std::sqrt(m2) : 0.0;
396 min_mass = std::min(min_mass, mass);
397 max_mass = std::max(max_mass, mass);
398
399 // Asymmetry
400 double asymmetry = std::fabs(pe_mag - pp_mag) / (pe_mag + pp_mag);
401 min_asym = std::min(min_asym, asymmetry);
402 max_asym = std::max(max_asym, asymmetry);
403 }
404 }
405
406 // Store results (use -9999 if no pairs found)
407 bool has_pairs = (!electrons.empty() && !positrons.empty());
408
409 bus_.set("min_dTanLambda", has_pairs ? min_dTanLambda : -9999.0);
410 bus_.set("max_dTanLambda", has_pairs ? max_dTanLambda : -9999.0);
411 bus_.set("min_opening", has_pairs ? min_opening : -9999.0);
412 bus_.set("max_opening", has_pairs ? max_opening : -9999.0);
413 bus_.set("min_mass", has_pairs ? min_mass : -9999.0);
414 bus_.set("max_mass", has_pairs ? max_mass : -9999.0);
415 bus_.set("min_asym", has_pairs ? min_asym : -9999.0);
416 bus_.set("max_asym", has_pairs ? max_asym : -9999.0);
417 bus_.set("min_dPhi", has_pairs ? min_dPhi : -9999.0);
418 bus_.set("max_dPhi", has_pairs ? max_dPhi : -9999.0);
419}

591bool PreselectAndCategorize2021::process(IEvent*) {
592 const auto& tsbank{bus_.get<TSData>("TSBank")};
593 const auto& eh{bus_.get<EventHeader>("EventHeader")};
594 int run_number = eh.getRunNumber();
595 if (biasingTool_) biasingTool_->setRun(run_number);
596
597 event_cf_.begin_event();
598
599 // re-apply trigger desicion to data sample since it contains
600 // other triggers within it
601 // MC is created with only this trigger AND the event header
602 // is not updated so we need to skip this check for MC
603 event_cf_.apply("single_trigger", (tsbank.isSingle3Trigger() || tsbank.isSingle2Trigger()));
604 event_cf_.fill_nm1("single_trigger", (tsbank.isSingle3Trigger() || tsbank.isSingle2Trigger()) ? 1 : 0);
605 if (not event_cf_.keep()) {
606 // we leave BEFORE filling the vertex counting histogram
607 // so that the vertex count histogram is relative to this trigger
608 // and does not include other triggers
609 return true;
610 }
611
612 const auto& allClusters{bus_.get<std::vector<CalCluster*>>("RecoEcalClusters")};
613
614 const auto& vtxs{bus_.get<std::vector<Vertex*>>(vtxColl_)};
628 std::vector<std::tuple<Vertex, Particle, Particle, double, double>> preselected_vtx;
629 int ivtx = 0;
630
631 for (Vertex* vtx : vtxs) {
632 // access the indiviual Vertex, electron, and positron
633 // and add corrections to them before applying pre-selection
634 int i_ele{-1}, i_pos{-1};
635 for (int ipart = 0; ipart < vtx->getParticles().GetEntries(); ++ipart) {
636 int pdg_id = ((Particle*)vtx->getParticles().At(ipart))->getPDG();
637 if (pdg_id == 11) {
638 i_ele = ipart;
639 } else if (pdg_id == -11) {
640 i_pos = ipart;
641 }
642 }
643 if (i_ele < 0 or i_pos < 0) {
644 throw std::runtime_error("Vertex formed without either an electron or positron!");
645 }
646
647 Particle ele = *dynamic_cast<Particle*>(vtx->getParticles().At(i_ele));
648 Particle pos = *dynamic_cast<Particle*>(vtx->getParticles().At(i_pos));
649
650 // for both particles, get cluster, update, re-set cluster
651 CalCluster ele_clu = ele.getCluster();
652 CalCluster pos_clu = pos.getCluster();
653
654 ele_clu.setTime(ele_clu.getTime() - calTimeOffset_);
655 pos_clu.setTime(pos_clu.getTime() - calTimeOffset_);
656
657 ele.setCluster(&ele_clu);
658 pos.setCluster(&pos_clu);
659
660 Track ele_trk = ele.getTrack();
661 Track pos_trk = pos.getTrack();
662
663 // cache z0 before any corrections (used to recompute isolation significance)
664 double ele_z0_raw = ele_trk.getZ0();
665 double pos_z0_raw = pos_trk.getZ0();
666
667 // snapshot pre-vertex-fit track momenta before they are overwritten
668 TVector3 ele_p_prefit(ele_trk.getMomentum()[0], ele_trk.getMomentum()[1], ele_trk.getMomentum()[2]);
669 TVector3 pos_p_prefit(pos_trk.getMomentum()[0], pos_trk.getMomentum()[1], pos_trk.getMomentum()[2]);
670
671 // replace particle track momenta with vertex-fitted momenta
672 bool is_top_ele = ele_trk.getTanLambda() > 0;
673 if (is_top_ele) {
674 if (vtx->getP1Y() > 0) {
675 ele_trk.setMomentum(vtx->getP1X(), vtx->getP1Y(), vtx->getP1Z());
676 pos_trk.setMomentum(vtx->getP2X(), vtx->getP2Y(), vtx->getP2Z());
677 } else {
678 ele_trk.setMomentum(vtx->getP2X(), vtx->getP2Y(), vtx->getP2Z());
679 pos_trk.setMomentum(vtx->getP1X(), vtx->getP1Y(), vtx->getP1Z());
680 }
681 }
682
683 // apply track_z0 and track_time corrections loaded from JSON
684 for (const auto& [name, corr] : track_corrections_) {
685 ele_trk.applyCorrection(name, corr);
686 pos_trk.applyCorrection(name, corr);
687 }
688
689 if (not v0proj_fits_.empty()) {
690 int run = eh.getRunNumber();
691 int closest_run;
692 closest_run = std::stoi((*v0proj_fits_.items().begin()).key());
693 for (auto entry : v0proj_fits_.items()) {
694 int check_run = std::stoi(entry.key());
695 if (check_run > run)
696 break;
697 else {
698 closest_run = check_run;
699 }
700 }
701 double elez0Mean = v0proj_fits_[std::to_string(closest_run)]["elez0_mean"];
702 double posz0Mean = v0proj_fits_[std::to_string(closest_run)]["posz0_mean"];
703
704 if (doZ0Corrections_) {
705 ele_trk.applyCorrection("z0", elez0Mean);
706 pos_trk.applyCorrection("z0", posz0Mean);
707 }
708 }
709
710 // Apply linear z0 calibration: z0 -= slope*tanL + intercept
711 if (biasingTool_) {
712 biasingTool_->updateWithCalibrateZ0(ele_trk);
713 biasingTool_->updateWithCalibrateZ0(pos_trk);
714 }
715
716 // Apply track smearing (z0 and momentum)
717 double ele_p_smear_ratio = 1.0;
718 double pos_p_smear_ratio = 1.0;
719 bool ele_has_truth_link = false;
720 bool pos_has_truth_link = false;
721
722 // Check for truth links using utility function (independent of smearing)
723 const std::vector<MCParticle*>* mcParticles = nullptr;
724 if (bus_.has(mcColl_)) {
725 mcParticles = &bus_.get<std::vector<MCParticle*>>(mcColl_);
726 ele_has_truth_link = utils::hasTruthMatch(ele_trk, mcParticles, debug_);
727 pos_has_truth_link = utils::hasTruthMatch(pos_trk, mcParticles, debug_);
728 }
729
730 // Apply smearing/corrections if tool is configured.
731 // doSmearing_ gates Gaussian smearing (MC only). On data, scale corrections
732 // always apply when the tool is loaded — the tool never adds Gaussian noise in data mode.
733 if (smearingTool_) {
734 // Set MC particles for smearing tool (needed for requireTruthMatch option)
735 if (mcParticles) {
736 smearingTool_->setMCParticles(mcParticles);
737 }
738 if (doSmearing_ || isData_) {
739 double ele_p_before = ele_trk.getP();
740 double pos_p_before = pos_trk.getP();
741 double psum_before = ele_p_before + pos_p_before;
742
743 // Apply z0 smearing first (if z0 corrections are enabled)
744 if (doZ0Corrections_) {
745 smearingTool_->updateWithSmearZ0(ele_trk);
746 smearingTool_->updateWithSmearZ0(pos_trk);
747 }
748
749 // Apply momentum smearing (omega or p)
750 if (smearOmega_) {
751 ele_p_smear_ratio = smearingTool_->updateWithSmearOmega(ele_trk);
752 pos_p_smear_ratio = smearingTool_->updateWithSmearOmega(pos_trk);
753 } else {
754 ele_p_smear_ratio = smearingTool_->updateWithSmearP(ele_trk);
755 pos_p_smear_ratio = smearingTool_->updateWithSmearP(pos_trk);
756 }
757 smearingTool_->updateVertexWithSmearP(vtx, ele_p_smear_ratio, pos_p_smear_ratio);
758
759 if (debug_) {
760 double ele_p_after = ele_trk.getP();
761 double pos_p_after = pos_trk.getP();
762 double psum_after = ele_p_after + pos_p_after;
763 std::cout << "[ScaleCorr] ele |p|: " << ele_p_before << " -> " << ele_p_after
764 << " ratio=" << ele_p_smear_ratio << std::endl;
765 std::cout << "[ScaleCorr] pos |p|: " << pos_p_before << " -> " << pos_p_after
766 << " ratio=" << pos_p_smear_ratio << std::endl;
767 std::cout << "[ScaleCorr] psum: " << psum_before << " -> " << psum_after
768 << " delta=" << (psum_after - psum_before) << std::endl;
769 }
770 }
771 }
772 bus_.set("ele_track_p", TVector3(ele_trk.getMomentum()[0], ele_trk.getMomentum()[1], ele_trk.getMomentum()[2]));
773 bus_.set("pos_track_p", TVector3(pos_trk.getMomentum()[0], pos_trk.getMomentum()[1], pos_trk.getMomentum()[2]));
774 bus_.set("ele_track_p_prefit", ele_p_prefit);
775 bus_.set("pos_track_p_prefit", pos_p_prefit);
776 bus_.set("ele_p_smear_ratio", ele_p_smear_ratio);
777 bus_.set("pos_p_smear_ratio", pos_p_smear_ratio);
778 bus_.set("ele_has_truth_link", ele_has_truth_link);
779 bus_.set("pos_has_truth_link", pos_has_truth_link);
780
781 // put tracks back into their particles
782 // with their new data
783 ele.setTrack(&ele_trk);
784 pos.setTrack(&pos_trk);
785
786 // corrections have been applied, now onto pre-selecting
787 int ele_nhits = ele.getTrack().getTrackerHitCount();
788 if (not ele.getTrack().isKalmanTrack()) ele_nhits *= 2;
789 int pos_nhits = pos.getTrack().getTrackerHitCount();
790 if (not pos.getTrack().isKalmanTrack()) pos_nhits *= 2;
791
792 if (debug_) {
793 std::cout << "Tracker hit counts (vertex " << ivtx << "):" << std::endl;
794 std::cout << " Electron: TrackerHitCount=" << ele.getTrack().getTrackerHitCount()
795 << ", nhits (after scaling)=" << ele_nhits << std::endl;
796 std::cout << " Positron: TrackerHitCount=" << pos.getTrack().getTrackerHitCount()
797 << ", nhits (after scaling)=" << pos_nhits << std::endl;
798 }
799
800 TVector3 ele_mom(ele.getTrack().getMomentum()[0], ele.getTrack().getMomentum()[1],
801 ele.getTrack().getMomentum()[2]);
802 TVector3 pos_mom(pos.getTrack().getMomentum()[0], pos.getTrack().getMomentum()[1],
803 pos.getTrack().getMomentum()[2]);
804
805 TVector3 psum = ele_mom + pos_mom;
806
807 vertex_cf_.begin_event();
808 {
809 const auto& posCl = pos.getCluster();
810 double posClE = posCl.getEnergy();
811 bool posClCut = posClE >= 0.2;
812
813 double pos_cl_min_dr = -9999.0;
814 double pos_cl_min_dr_t = -9999.0;
815 if (posClE < -9000.0) {
816 auto posAtEcal = pos_trk.getPositionAtEcal();
817 double min_dr = 9999.0;
818 for (const auto* cl : allClusters) {
819 auto clPos = cl->getPosition();
820 double dx = clPos[0] - posAtEcal[0];
821 double dy = clPos[1] - posAtEcal[1];
822 double dr = std::sqrt(dx*dx + dy*dy);
823 if (dr < min_dr) {
824 min_dr = dr;
825 pos_cl_min_dr_t = cl->getTime() - calTimeOffset_;
826 }
827 }
828 pos_cl_min_dr = min_dr;
829 }
830 bus_.set("pos_cl_min_dr", pos_cl_min_dr);
831 bus_.set("pos_cl_min_dr_t", pos_cl_min_dr_t);
832
833 vertex_cf_.apply("positron_clusterE_above_0pt2GeV", posClCut);
834 }
835 vertex_cf_.apply("ele_track_cluster",
836 disableTimingCuts_ || fabs(ele.getTrack().getTrackTime() - pos.getCluster().getTime()) <= time_cuts_[0]);
837 vertex_cf_.apply("pos_track_cluster",
838 disableTimingCuts_ || fabs(pos.getTrack().getTrackTime() - pos.getCluster().getTime()) <= time_cuts_[1]);
839 vertex_cf_.apply("ele_pos_track",
840 disableTimingCuts_ || fabs(ele.getTrack().getTrackTime() - pos.getTrack().getTrackTime()) <= time_cuts_[2]);
841 vertex_cf_.apply("ele_track_chi2ndf", ele.getTrack().getChi2Ndf() <= 20.0);
842 vertex_cf_.apply("pos_track_chi2ndf", pos.getTrack().getChi2Ndf() <= 20.0);
843 vertex_cf_.apply("electron_below_2pt9GeV", ele.getTrack().getP() <= 2.9);
844 vertex_cf_.apply("electron_above_0pt4GeV", ele.getTrack().getP() >= 0.4);
845 vertex_cf_.apply("positron_above_0pt4GeV", pos.getTrack().getP() >= 0.4);
846 vertex_cf_.apply("ele_min_hits", ele_nhits >= minHits_);
847 vertex_cf_.apply("pos_min_hits", pos_nhits >= minHits_);
848 vertex_cf_.apply("vertex_chi2", vtx->getChi2() <= 20.0);
849 double vtxmaxp = ele_mom.Mag() + pos_mom.Mag();
850 vertex_cf_.apply("vtx_max_p_4pt0GeV", psum.Mag() <= 4.0);
851
852 vertex_cf_.fill_nm1("positron_clusterE_above_0pt2GeV", pos.getCluster().getEnergy());
853 vertex_cf_.fill_nm1("ele_track_cluster", fabs(ele.getTrack().getTrackTime() - pos.getCluster().getTime()));
854 vertex_cf_.fill_nm1("pos_track_cluster", fabs(pos.getTrack().getTrackTime() - pos.getCluster().getTime()));
855 vertex_cf_.fill_nm1("ele_pos_track", fabs(ele.getTrack().getTrackTime() - pos.getTrack().getTrackTime()));
856 vertex_cf_.fill_nm1("ele_track_chi2ndf", ele.getTrack().getChi2Ndf());
857 vertex_cf_.fill_nm1("pos_track_chi2ndf", pos.getTrack().getChi2Ndf());
858 vertex_cf_.fill_nm1("electron_below_2pt9GeV", ele.getTrack().getP());
859 vertex_cf_.fill_nm1("electron_above_0pt4GeV", ele.getTrack().getP());
860 vertex_cf_.fill_nm1("positron_above_0pt4GeV", pos.getTrack().getP());
861 vertex_cf_.fill_nm1("ele_min_hits", ele_nhits);
862 vertex_cf_.fill_nm1("pos_min_hits", pos_nhits);
863 vertex_cf_.fill_nm1("vertex_chi2", vtx->getChi2());
864 vertex_cf_.fill_nm1("vtx_max_p_4pt0GeV", vtxmaxp);
865
866 if (disablePreselection_ || vertex_cf_.keep()) {
867 preselected_vtx.emplace_back(*vtx, ele, pos, ele_z0_raw, pos_z0_raw);
868 }
869 ivtx++;
870 }
871
872 n_vertices_h_->Fill(vtxs.size(), preselected_vtx.size());
873 event_cf_.apply("at_least_one_vertex", disablePreselection_ || preselected_vtx.size() >= 1);
874 event_cf_.apply("no_extra_vertices", disablePreselection_ || preselected_vtx.size() < 2);
875
876 event_cf_.fill_nm1("at_least_one_vertex", preselected_vtx.size());
877 event_cf_.fill_nm1("no_extra_vertices", preselected_vtx.size());
878 if (not event_cf_.keep()) {
879 return true;
880 }
881
882 bool single2{false}, single3{false};
883 if (tsbank.isSingle3Trigger() == 1) single3 = true;
884 if (tsbank.isSingle2Trigger() == 1) single2 = true;
885 bus_.set("single2", single2);
886 bus_.set("single3", single3);
887
888 // correct number of vertices (i.e. only one)
889 // unpack the vector of vertices into the single elements
890 // Guard against empty collection (can occur when disablePreselection=true bypasses at_least_one_vertex)
891 if (preselected_vtx.empty()) return true;
892 auto [vtx, ele, pos, ele_z0_raw, pos_z0_raw] = preselected_vtx.at(0);
893
894 // earliest layer hit categories
895 bool eleL1{false}, eleL2{false}, eleL3{false}, eleL4{false};
896 bool posL1{false}, posL2{false}, posL3{false}, posL4{false};
897 Track ele_trk{ele.getTrack()}, pos_trk{pos.getTrack()};
898 auto ele_layers = _ah->GetTrackHitLayers(&ele_trk);
899 auto pos_layers = _ah->GetTrackHitLayers(&pos_trk);
900
901 if (ele_layers.at(0) == 1 && ele_layers.at(1) == 1) eleL1 = true;
902 if (ele_layers.at(2) == 1 && ele_layers.at(3) == 1) eleL2 = true;
903 if (ele_layers.at(4) == 1 && ele_layers.at(5) == 1) eleL3 = true;
904 if (ele_layers.at(6) == 1 && ele_layers.at(7) == 1) eleL4 = true;
905 if (pos_layers.at(0) == 1 && pos_layers.at(1) == 1) posL1 = true;
906 if (pos_layers.at(2) == 1 && pos_layers.at(3) == 1) posL2 = true;
907 if (pos_layers.at(4) == 1 && pos_layers.at(5) == 1) posL3 = true;
908 if (pos_layers.at(6) == 1 && pos_layers.at(7) == 1) posL4 = true;
909
910 bus_.set("eleL1", eleL1);
911 bus_.set("eleL2", eleL2);
912 bus_.set("eleL3", eleL3);
913 bus_.set("eleL4", eleL4);
914 bus_.set("posL1", posL1);
915 bus_.set("posL2", posL2);
916 bus_.set("posL3", posL3);
917 bus_.set("posL4", posL4);
918
919 // vertex layer category flags (outward hit requirement: LN track requires L[N..3] hits and no L[1..N-1])
920 bool ele_L1 = eleL1 && eleL2 && eleL3;
921 bool ele_L2 = !eleL1 && eleL2 && eleL3;
922 bool ele_L3 = !eleL1 && !eleL2 && eleL3;
923 bool pos_L1 = posL1 && posL2 && posL3;
924 bool pos_L2 = !posL1 && posL2 && posL3;
925 bool pos_L3 = !posL1 && !posL2 && posL3;
926
927 bus_.set("isL1L1", ele_L1 && pos_L1);
928 bus_.set("isL2L2", ele_L2 && pos_L2);
929 bus_.set("isL3L3", ele_L3 && pos_L3);
930 bus_.set("isL1L2", (ele_L1 && pos_L2) || (pos_L1 && ele_L2));
931 bus_.set("isL2L3", (ele_L2 && pos_L3) || (pos_L2 && ele_L3));
932
933 // top/bottom identification
934 bool ele_isTop = ele_trk.getTanLambda() > 0;
935 bool pos_isTop = pos_trk.getTanLambda() > 0;
936 bus_.set("ele_isTop", ele_isTop);
937 bus_.set("pos_isTop", pos_isTop);
938
939 // Per-layer axial/stereo hit flags
940 // From sensor_locations.txt z-positions:
941 // Top: even index (0,2,4,6) = axial, odd index (1,3,5,7) = stereo
942 // Bottom: even index (0,2,4,6) = stereo, odd index (1,3,5,7) = axial
943 auto setLayerHits = [&](const std::vector<int>& layers, bool isTop, const std::string& prefix) {
944 for (int iL = 0; iL < 4; iL++) {
945 int idx_even = 2 * iL;
946 int idx_odd = 2 * iL + 1;
947 bool hit_axial = isTop ? (layers.at(idx_even) == 1) : (layers.at(idx_odd) == 1);
948 bool hit_stereo = isTop ? (layers.at(idx_odd) == 1) : (layers.at(idx_even) == 1);
949 bus_.set(prefix + "L" + std::to_string(iL + 1) + "_axial", hit_axial);
950 bus_.set(prefix + "L" + std::to_string(iL + 1) + "_stereo", hit_stereo);
951 }
952 };
953 setLayerHits(ele_layers, ele_isTop, "ele");
954 setLayerHits(pos_layers, pos_isTop, "pos");
955
956 double ele_L1_iso{9999.0}, pos_L1_iso{9999.0};
957 if (eleL1 && eleL2 && posL1 && posL2) {
958 ele_L1_iso = ele_trk.getIsolation(0);
959 pos_L1_iso = pos_trk.getIsolation(0);
960 }
961 bus_.set("ele_L1_iso", ele_L1_iso);
962 bus_.set("pos_L1_iso", pos_L1_iso);
963
964 // isolation significance
965 double ele_L1_iso_significance{9999.0}, pos_L1_iso_significance{9999.0};
966
