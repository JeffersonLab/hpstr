59class Cutflow {
60
61 public:
62 Cutflow(const std::string& name, const std::string& nocut_name):
63 name_{name}, nocut_name_{nocut_name} {}
64
65 void set_label_names(const std::vector<std::string>& label_names) {
66 label_names_ = label_names;
67 }
68
69 void add(const std::string& name, int nbins, float min, float max) {
70 if (h_cutflow_) {
71 throw std::runtime_error("Cannot add more cuts after init");
72 }
73
74 auto h = std::make_unique<TH1F>(
75 ("nm1_" + name + "_h").c_str(),
76 (name + " N-1").c_str(),
77 nbins, min, max);
78
79 cuts_.emplace(name, std::make_pair<std::size_t, std::unique_ptr<TH1F>>(cuts_.size(), std::move(h)));
80 }
81
82 void init() {
83 if (h_cutflow_) {
84 throw std::runtime_error("Cannot init twice.");
85 }
86
87 h_cutflow_ = std::make_unique<TH1F>(
88 (name_ + "_cutflow_h").c_str(),
89 (name_ + " cutflow;;N_{events}").c_str(),
90 cuts_.size()+1, -1.5, cuts_.size()-0.5
91 );
92
93 cut_desc_.resize(cuts_.size());
94 }
95
96 void begin_event() {
97 h_cutflow_->Fill(nocut_name_.c_str(), 1.);
98 keep_ = true;
99 for (auto& d : cut_desc_) d.reset();
100 }
101
102 void apply(const std::string& name, bool descision) {
103 if (cuts_.find(name) == cuts_.end()) {
104 throw std::runtime_error("Cut "+name+" not `add`ed to cutflow.");
105 }
106
107 keep_ = (keep_ and descision);
108 if (keep_) h_cutflow_->Fill(name.c_str(), 1.);
109 int i_cut = cuts_.at(name).first;
110 cut_desc_[i_cut] = descision;
111 }
112
113 template<typename T>
114 void fill_nm1(const std::string& name, T value) {
115 bool should_fill{true};
116
117 for (std::size_t i_cut{0}; i_cut < cuts_.size(); i_cut++) {
118 if (i_cut != cuts_[name].first and cut_desc_[i_cut].has_value()) {
119 should_fill = (should_fill and cut_desc_[i_cut].value());
120 }
121 }
122
123 if (should_fill) {
124 cuts_.at(name).second->Fill(value);
125 }
126 }
127
128 bool keep() const {
129 return keep_;
130 }
131
132 void save() {
133 // if label names not set or size mismatch, just skip setting labels
134 for (int i = 0; i < h_cutflow_->GetNbinsX(); ++i) {
135 if (label_names_.size() == (std::size_t)h_cutflow_->GetNbinsX()) {
136 h_cutflow_->GetXaxis()->SetBinLabel(i+1, label_names_.at(i).c_str());
137 }
138 }
139 h_cutflow_->Write();
140 for (const auto& [_name, entry] : cuts_) entry.second->Write();
141 }
142
143 private:
144 std::string name_, nocut_name_;
145 std::unique_ptr<TH1F> h_cutflow_;
146 std::map<std::string,std::pair<std::size_t,std::unique_ptr<TH1F>>> cuts_;
147 bool keep_;
148 std::vector<std::optional<bool>> cut_desc_;
149 std::vector<std::string> label_names_;
150};

160 template<typename BaggageType>
161 void board_input(TTree* tree, const std::string& name) {
162 if (bus_.isOnBoard(name)) {
163 throw std::runtime_error(name + " is already on board the bus.");
164 }
165
166 bus_.board<BaggageType>(name);
167 if (bus_.attach(tree, name, false /* create if missing */) == 0) {
168 throw std::runtime_error("Unable to find branch " + name + " in input TTree.");
169 }
170 }
171
172 template<typename BaggageType>
173 void board_output(TTree* tree, const std::string& name) {
174 if (bus_.isOnBoard(name)) {
175 throw std::runtime_error(name + " is already on board the bus.");
176 }
177 bus_.board<BaggageType>(name);
178 if (bus_.attach(tree, name, true /* create if missing */) == 0) {
179 throw std::runtime_error("Unable to create branch " + name + " in output TTree.");
180 }
181 }
182
183 template<typename BaggageType>
184 const BaggageType& get(const std::string& name) {
185 return bus_.get<BaggageType>(name);
186 }
187
188 template<typename BaggageType>
189 void set(const std::string& name, const BaggageType& obj) {
190 if (bus_.isOnBoard(name)) {
191 bus_.update(name, obj);
192 return;
193 }
194 throw std::runtime_error("No branch " + name + " boarded on the bus.");
195 }
