307bool FeeAnaProcessor::process(IEvent* ievent) {
308
309 double weight = 1.;
310 int n_sel_tracks = 0;
311
312 double minTime = isData_ ? clusterTimeMin_ : clusterTimeMinMC_;
313 double maxTime = isData_ ? clusterTimeMax_ : clusterTimeMaxMC_;
314 double calOffset = isData_ ? calTimeOffset_ : calTimeOffsetMC_;
315
316 if (debug_ > 0) {
317 std::cout << "[FeeAna] --- new event isData=" << isData_
318 << " feeEnergyMin=" << feeClusterEnergyMin_
319 << " timeWindow=[" << minTime << "," << maxTime << "]"
320 << " nFSPs=" << fsps_->size() << std::endl;
321 }
322
323 // Require the FEE trigger on data. MC is generated with a single trigger and the
324 // TSBank bits are not filled, so the requirement is applied to data only (mirrors
325 // the trigger handling in PreselectAndCategorize2021).
326 if (requireFeeTrigger_ && isData_) {
327 if (!tsdata_ || !tsdata_->isFEETrigger()) {
328 if (debug_ > 0) std::cout << "[FeeAna] FAIL FEE trigger" << std::endl;
329 return true;
330 }
331 }
332
333 // FEE selection: loop over FinalStateParticles. Each particle carries its track and
334 // (if matched at the LCIO stage) its ECal cluster, so no manual track-cluster matching
335 // is needed. We require an electron whose associated cluster looks like a full-energy
336 // electron (energy / time / E-over-p), then apply the usual track-quality cuts.
337 for (unsigned int ipart = 0; ipart < fsps_->size(); ++ipart) {
338
339 if (trkSelector_) trkSelector_->getCutFlowHisto()->Fill(0.,weight);
340
341 Particle* part = fsps_->at(ipart);
342
343 // Electron requirement (FEEs are electrons; charge < 0 in HPS convention)
344 if (requireElectron_ && part->getCharge() >= 0) {
345 if (debug_ > 0) std::cout << "[FeeAna] FAIL not an electron: charge=" << part->getCharge() << std::endl;
346 continue;
347 }
348
349 // Associated FEE cluster (already matched to the track at the LCIO stage).
350 // A particle with no associated cluster has the sentinel cluster energy.
351 CalCluster clu = part->getCluster();
352 double clu_E = clu.getEnergy();
353 double clu_traw = clu.getTime(); // raw cluster time
354 double clu_t = clu_traw - calOffset; // offset-corrected (centered near 0)
355
356 // DIAGNOSTIC: dump cluster time for every electron with a full-energy cluster,
357 // *before* the time cut, to verify the CalTimeOffset centers the prompt peak.
358 if (debug_ > 0 && part->getCharge() < 0 && clu_E > feeClusterEnergyMin_) {
359 std::cout << "[FeeAna][cluTime] rawCluTime=" << clu_traw
360 << " corrCluTime=" << clu_t
361 << " cluE=" << clu_E
362 << " trkTime=" << part->getTrack().getTrackTime()
363 << std::endl;
364 }
365
366 if (requireCluster_) {
367 if (clu_E < feeClusterEnergyMin_) {
368 if (debug_ > 0) std::cout << "[FeeAna] FAIL cluster E: " << clu_E << " < " << feeClusterEnergyMin_ << std::endl;
369 continue;
370 }
371 if (clu_t < minTime || clu_t > maxTime) {
372 if (debug_ > 0) std::cout << "[FeeAna] FAIL cluster time (corrected): " << clu_t << " not in [" << minTime << "," << maxTime << "]" << std::endl;
373 continue;
374 }
375 }
376
377 // Track associated to this particle (returned by value)
378 Track track_obj = part->getTrack();
379 Track* track = &track_obj;
380 int n2dhits_onTrack = !track->isKalmanTrack() ? track->getTrackerHitCount() * 2 : track->getTrackerHitCount();
381
382 // E/p of the FEE candidate
383 double eop = track->getP() > 0 ? clu_E / track->getP() : -1.0;
384 if (requireCluster_ && (eop < eopMin_ || eop > eopMax_)) {
385 if (debug_ > 0) std::cout << "[FeeAna] FAIL E/p: " << eop << " not in [" << eopMin_ << "," << eopMax_ << "]" << std::endl;
386 continue;
387 }
388
389 TVector3 trk_mom;
390 trk_mom.SetX(track->getMomentum()[0]);
391 trk_mom.SetY(track->getMomentum()[1]);
392 trk_mom.SetZ(track->getMomentum()[2]);
393
394 // Decode per-sensor hit layer vector and determine top/bottom.
395 bool isTop = track->getTanLambda() > 0;
396 // The FEE track comes from the FinalStateParticle, whose embedded Track does NOT
397 // carry the per-sensor hit-layer vector: FinalStateParticleProcessor fills the hit
398 // count and the SvtHit refs but never calls addHitLayer(), so getHitLayers() is
399 // empty here. The matching KalmanFullTracks entry (same track ID) does carry it, so
400 // match by ID and decode the layers from that. Fall back to the FSP track's own
401 // SvtHits if no match is found, and finally to the (empty) FSP layer vector.
402 Track* layerTrack = track;
403 if (allTracks_) {
404 for (Track* t : *allTracks_) {
405 if (t && t->getID() == track->getID()) { layerTrack = t; break; }
406 }
407 }
408 std::vector<int> layers = ah_->GetTrackHitLayers(layerTrack);
409 if (std::none_of(layers.begin(), layers.end(), [](int v){ return v == 1; })) {
410 // Matched full track had no layer vector; decode from this track's SvtHits.
411 TRefArray hits = track->getSvtHits();
412 for (int ih = 0; ih < hits.GetEntries(); ++ih) {
413 TrackerHit* th = (TrackerHit*) hits.At(ih);
414 if (th && th->getLayer() >= 0 && th->getLayer() < (int) layers.size())
415 layers.at(th->getLayer()) = 1;
416 }
417 }
418 // Encode the full per-sensor hit pattern as a bitmask (bit i = sensor slot i hit).
419 int hit_pattern = 0;
420 for (int i = 0; i < (int) layers.size(); ++i)
421 if (layers.at(i) == 1) hit_pattern |= (1 << i);
422
423 // Per-hit_pattern momentum scale correction (data momentum -> beam energy).
424 // p_corr = p * scale, keyed by hit_pattern (default_scale for unlisted patterns).
425 double hitPatternScale = 1.0;
426 bool hitPatternScaleIsDefault = false;
427 if (doHitPatternScale_ && (!hitPatternScaleDataOnly_ || isData_)) {
428 auto it = hitPatternScale_.find(hit_pattern);
429 if (it != hitPatternScale_.end()) {
430 hitPatternScale = it->second;
431 } else {
432 hitPatternScale = hitPatternDefaultScale_;
433 hitPatternScaleIsDefault = true;
434 }
435 }
436 // Corrected track: scale the momentum magnitude by hitPatternScale (omega scales as
437 // 1/scale since p is inversely proportional to |omega|).
438 Track trk_corr = *track;
439 if (hitPatternScale != 1.0) {
440 std::vector<double> mom = trk_corr.getMomentum();
441 for (double& c : mom) c *= hitPatternScale;
442 trk_corr.setMomentum(mom);
443 trk_corr.setOmega(trk_corr.getOmega() / hitPatternScale);
444 }
445
446 // Convention from sensor_locations.txt:
447 // Top: even sensor index = axial, odd = stereo
448 // Bottom: even sensor index = stereo, odd = axial
449 bool L1_axial = isTop ? (layers.at(0) == 1) : (layers.at(1) == 1);

452 bool L2_stereo = isTop ? (layers.at(3) == 1) : (layers.at(2) == 1);
453 bool L3_axial = isTop ? (layers.at(4) == 1) : (layers.at(5) == 1);
454 bool L3_stereo = isTop ? (layers.at(5) == 1) : (layers.at(4) == 1);
455 bool L4_axial = isTop ? (layers.at(6) == 1) : (layers.at(7) == 1);
456 bool L4_stereo = isTop ? (layers.at(7) == 1) : (layers.at(6) == 1);

461 bool L2 = (layers.at(2) == 1 && layers.at(3) == 1);
462 bool L3 = (layers.at(4) == 1 && layers.at(5) == 1);
463 bool L4 = (layers.at(6) == 1 && layers.at(7) == 1);

466 // ele_L1/ele_L2/ele_L3 vertex categories in PreselectAndCategorize2021. For a single
467 // FEE track these are the per-track analog of the isL1L1/isL2L2/isL3L3 vertex flags:
468 // is_L1 = track starts at L1, is_L2 = starts at L2, is_L3 = starts at L3.
469 bool is_L1 = L1 && L2 && L3;
470 bool is_L2 = !L1 && L2 && L3;
471 bool is_L3 = !L1 && !L2 && L3;

496 continue;

534 // compared to the offset-corrected cluster time, using the same MC track-time

537 int n_tracks_intime = 0;

539 if (ecalClusters_) {

541 if (!c || c->getEnergy() < vetoClusterEnergyMin_) continue;
542 if (std::fabs(c->getTime() - clu_traw) < vetoClusterTimeWindow_) ++n_clusters_intime;

544 }
545 if (allTracks_) {

592 trkHistos_->Fill2DTrack(track);

596 truthHistos_->Fill2DTrack(track);
597 truthHistos_->Fill1DTrackTruth(track, truth_track);

605 track->getPositionAtEcal()[1]);

609 track->getTanLambda(),
610 track->getP());

613 track->getTrackerHitCount(),
614 track->getP());

623 track->getP());

731 L2_stereo_out_ = L2_stereo;
732 L3_axial_out_ = L3_axial;
733 L3_stereo_out_ = L3_stereo;
734 L4_axial_out_ = L4_axial;

737 L2_out_ = L2;
738 L3_out_ = L3;
739 L4_out_ = L4;
740 is_L1_out_ = is_L1;
741 is_L2_out_ = is_L2;
742 is_L3_out_ = is_L3;
743 hit_pattern_out_ = hit_pattern;

752
753 }//Loop on FinalStateParticles

755 trkHistos_->Fill1DHisto("n_tracks_h",n_sel_tracks);
