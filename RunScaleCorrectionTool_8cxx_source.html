21// Same lookup as TrackSmearingTool::lookupBinnedValue: clamp to the first/last bin.
22double lookupBinned(const std::vector<double>& edges, const std::vector<double>& values, double x) {
23 if (edges.size() < 2 || values.empty()) return 0.;
24 if (x <= edges.front()) return values.front();
25 if (x >= edges.back()) return values.back();
26 int bin = static_cast<int>(std::distance(edges.begin(), std::upper_bound(edges.begin(), edges.end(), x))) - 1;
27 bin = std::max(0, std::min(bin, static_cast<int>(values.size()) - 1));
28 return values[bin];
29}

91double RunScaleCorrectionTool::omegaMeanShift(const Track& track, bool isTop) const {
92 if (binnedMeans_) {
93 const double x = lookupValue(track);
94 return isTop ? lookupBinned(edgesTop_, muMcTop_, x) - lookupBinned(edgesTop_, muDataTop_, x)
95 : lookupBinned(edgesBot_, muMcBot_, x) - lookupBinned(edgesBot_, muDataBot_, x);
96 }
97 return isTop ? (omegaMuMcTop_ - omegaMuDataTop_) : (omegaMuMcBot_ - omegaMuDataBot_);

110double RunScaleCorrectionTool::apply(Track& track, int run) const {
111 const bool isTop = track.getTanLambda() > 0;
112 const double omega = track.getOmega();
113 double scale = getScale(run, isTop);
114 double omegaCorr = omega / scale;
115
116 if (!meanCorrCfg_.empty()) {
117 // Same as TrackSmearingTool's data-mode omega mean correction
118 omegaCorr += omegaMeanShift(track, isTop);
119 if (omegaCorr != 0.) scale = omega / omegaCorr;
120 }
121
122 if (scale == 1.0) return scale;
123 std::vector<double> mom = track.getMomentum();
124 for (double& c : mom) c *= scale;
125 track.setMomentum(mom);
126 track.setOmega(omega / scale);
127 return scale;
128}
