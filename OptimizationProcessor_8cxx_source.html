124double OptimizationProcessor::calculateZBi(double n_on, double n_off, double tau) {
125 double P_Bi = TMath::BetaIncomplete(1. / (1. + tau), n_on, n_off + 1);
126 // std::cout << "P_Bi: " << P_Bi << std::endl;
127 double Z_Bi = std::pow(2, 0.5) * TMath::ErfInverse(1 - 2 * P_Bi);
128 // std::cout << "Z_Bi: " << Z_Bi << std::endl;
129 return Z_Bi;
130}
