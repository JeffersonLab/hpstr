36void ClusterEfficiency2021::initialize(TTree* tree) {
37 _ah = std::make_shared<AnaHelpers>();
38
39 bus_.board_input<EventHeader>(tree, "EventHeader");
40 bus_.board_input<TSData>(tree, "TSBank");
41 if (not trkColl_.empty()) bus_.board_input<std::vector<Track*>>(tree, trkColl_);
42 bus_.board_input<std::vector<Vertex*>>(tree, vtxColl_);
43 bus_.board_input<std::vector<CalCluster*>>(tree, "RecoEcalClusters");
44 if (not isData_ and not mcColl_.empty()) bus_.board_input<std::vector<MCParticle*>>(tree, mcColl_);
45
46 // efficiency chain (quick at-a-glance, in addition to the tree)
47 event_cf_.add("single_trigger", 2, -0.5, 1.5);
48 event_cf_.add("truth_epem_found", 2, -0.5, 1.5);
49 event_cf_.add("has_v0", 2, -0.5, 1.5);
50 event_cf_.add("both_tracks_fiducial", 2, -0.5, 1.5);
51 event_cf_.add("pos_has_cluster", 2, -0.5, 1.5);
52 event_cf_.init();
53 event_cf_.set_label_names({"readout", "single2 or 3 trigger", "truth e^{+}e^{-} found",
54 "N_{vtx} #geq 1", "both tracks fiducial", "e^{+} has cluster"});
55}

119bool ClusterEfficiency2021::process(IEvent*) {
120 const auto& tsbank{bus_.get<TSData>("TSBank")};
121 const auto& eh{bus_.get<EventHeader>("EventHeader")};
122
123 event_cf_.begin_event();
124
125 // Re-apply trigger decision: data contains other triggers; MC is produced with only
126 // this trigger but the event header is not updated, so the bits are still meaningful.
127 bool single2 = tsbank.isSingle2Trigger() == 1;
128 bool single3 = tsbank.isSingle3Trigger() == 1;
129 bool single_trigger = single2 || single3;
130 event_cf_.apply("single_trigger", single_trigger);
131 event_cf_.fill_nm1("single_trigger", single_trigger ? 1 : 0);
132 if (not event_cf_.keep()) return true;
133
134 // ===== Q1: after the singles2/3 trigger, is there an energetic cluster ANYWHERE? =====
135 // Loop over ALL RecoEcalClusters, independent of any v0. By definition a singles
136 // trigger requires an energetic cluster, so this should be ~100%.
137 const auto& allClusters{bus_.get<std::vector<CalCluster*>>("RecoEcalClusters")};
138 int n_clusters = static_cast<int>(allClusters.size());
139 double event_max_clu_E = -9999.;
140 for (const auto* cl : allClusters) {
141 if (cl->getEnergy() > event_max_clu_E) event_max_clu_E = cl->getEnergy();
142 }
143 bool event_has_cluster = event_max_clu_E > eventClusterThresh_;
144 n_triggered_++;
145 if (event_has_cluster) n_event_has_cluster_++;
146
147 // nearest ENERGETIC cluster (E > eventClusterThresh_) to a track's ECal projection.
148 // A track is "dr-matched" to a trigger-quality cluster if the nearest such cluster is
149 // within drMatch_. Used to score/pick the best v0 and to fill the per-track dbg below.
150 auto nearest_energetic = [&](const std::vector<double>& ecal,
151 double& out_dr, double& out_E, double& out_t) {
152 out_dr = 9999.0; out_E = -9999.0; out_t = -9999.0;
153 for (const auto* cl : allClusters) {
154 if (cl->getEnergy() <= eventClusterThresh_) continue;
155 auto cp = cl->getPosition();
156 double dr = std::sqrt((cp[0] - ecal[0]) * (cp[0] - ecal[0]) +
157 (cp[1] - ecal[1]) * (cp[1] - ecal[1]));
158 if (dr < out_dr) { out_dr = dr; out_E = cl->getEnergy(); out_t = cl->getTime() - calTimeOffset_; }
159 }
160 };
161
162 // ----- defaults (sentinels) -----
163 bus_.set("weight", 1.);
164 bus_.set("run", eh.getRunNumber());
165 bus_.set("event", eh.getEventNumber());
166 bus_.set("single2", single2);
167 bus_.set("single3", single3);
168
169 bus_.set("ele_truth_p", TVector3(0., 0., 0.));
170 bus_.set("pos_truth_p", TVector3(0., 0., 0.));
171 bus_.set("ele_truth_E", -9999.);
172 bus_.set("pos_truth_E", -9999.);
173 bus_.set("true_vertex_invM", -9999.);
174 bus_.set("true_vertex_psum", -9999.);
175 bus_.set("true_decay_len", -9999.);
176 bus_.set("true_betagamma", -9999.);
177
178 bus_.set("ele_track_p", TVector3(0., 0., 0.));
179 bus_.set("pos_track_p", TVector3(0., 0., 0.));
180 for (const auto& name : {"ele_ecal_x", "ele_ecal_y", "ele_ecal_z",
181 "pos_ecal_x", "pos_ecal_y", "pos_ecal_z",
182 "ele_clu_E", "ele_clu_x", "ele_clu_y", "ele_clu_time",
183 "pos_clu_E", "pos_clu_x", "pos_clu_y", "pos_clu_time",
184 "pos_nearest_clu_dr", "pos_nearest_clu_E", "pos_nearest_clu_time"}) {
185 bus_.set<double>(name, -9999.);
186 }
187 bus_.set("ele_clu_nhits", -1);
188 bus_.set("pos_clu_nhits", -1);
189
190 bool truth_epem_found = false, has_v0 = false;
191 bool ele_fiducial = false, pos_fiducial = false, both_tracks_fiducial = false;
192 bool ele_has_cluster = false, pos_has_cluster = false;
193 // Q2: is an energetic cluster matched to the v0 positron track?
194 bool pos_matched_cluster = false; // associated cluster with E > eventClusterThresh_
195 double pos_clu_E_dbg = -9999.; // associated-cluster energy (sentinel if none)
196 double pos_nearest_dr_dbg = -9999.; // dr to nearest standalone cluster at ECal
197 // Q3: dr-based track->cluster match (did the e-/e+ make an energetic trigger cluster?)
198 bool ele_dr_matched = false, pos_dr_matched = false;
199 double ele_nearest_dr_dbg = -9999., ele_dr_E_dbg = -9999.;
200 double pos_dr_E_dbg = -9999.;
201
202 // ===================== truth block =====================
203 const std::vector<MCParticle*>* mcParticles = nullptr;
204 if (bus_.has(mcColl_)) {
205 mcParticles = &bus_.get<std::vector<MCParticle*>>(mcColl_);
206 MCParticle* mother{nullptr};
207 ROOT::Math::PxPyPzEVector trueEleP, truePosP;
208 bool foundEle = false, foundPos = false;
209 for (MCParticle* ptr : *mcParticles) {
210 std::vector<double> lP = ptr->getMomentum();
211 if (ptr->getPDG() == apPDG_ || (isSimpSignal_ && ptr->getPDG() == 625)) {
212 mother = ptr;
213 } else if (ptr->getPDG() == 11 && ptr->getMomPDG() == signalMomPDG_) {
214 trueEleP = ROOT::Math::PxPyPzEVector(lP.at(0), lP.at(1), lP.at(2), ptr->getEnergy());
215 foundEle = true;
216 } else if (ptr->getPDG() == -11 && ptr->getMomPDG() == signalMomPDG_) {
217 truePosP = ROOT::Math::PxPyPzEVector(lP.at(0), lP.at(1), lP.at(2), ptr->getEnergy());
218 foundPos = true;
219 }
220 }
221 truth_epem_found = foundEle && foundPos;
222
223 if (foundEle) {
224 bus_.set("ele_truth_p", TVector3(trueEleP.Px(), trueEleP.Py(), trueEleP.Pz()));
225 bus_.set("ele_truth_E", trueEleP.E());
226 }
227 if (foundPos) {
228 bus_.set("pos_truth_p", TVector3(truePosP.Px(), truePosP.Py(), truePosP.Pz()));
229 bus_.set("pos_truth_E", truePosP.E());
230 }
231 if (foundEle && foundPos) {
232 bus_.set("true_vertex_invM", (trueEleP + truePosP).M());
233 bus_.set("true_vertex_psum", trueEleP.P() + truePosP.P());
234 }
235
236 if (mother) {
237 std::vector<double> vtxPos = mother->getVertexPosition();
238 std::vector<double> endPos = mother->getEndPoint();
239 double dx = endPos[0] - vtxPos[0];
240 double dy = endPos[1] - vtxPos[1];
241 double dz = endPos[2] - vtxPos[2];
242 bus_.set("true_decay_len", std::sqrt(dx * dx + dy * dy + dz * dz));
243
244 std::vector<double> mMom = mother->getMomentum();
245 double mp = std::sqrt(mMom[0] * mMom[0] + mMom[1] * mMom[1] + mMom[2] * mMom[2]);
246 if (mother->getMass() > 0) bus_.set("true_betagamma", mp / mother->getMass());
247 }
248 }
249 event_cf_.apply("truth_epem_found", isData_ ? true : truth_epem_found);
250 event_cf_.fill_nm1("truth_epem_found", truth_epem_found ? 1 : 0);
251
252 // ===================== reco block =====================
253 const auto& vtxs{bus_.get<std::vector<Vertex*>>(vtxColl_)};
254
255 // Evaluate EVERY e-/e+ v0 and choose the BEST one = the v0 whose positron points closest
256 // to an energetic cluster (smallest e+ dr). Also record whether ANY v0 has a matched
257 // e+/e- (best-case efficiency) and what the FIRST v0 alone would have given.
258 Vertex* chosen{nullptr};
259 int chosen_i_ele{-1}, chosen_i_pos{-1};
260 int n_v0 = 0;
261 double best_pos_dr = 1e9;
262 bool any_pos_matched = false, any_ele_matched = false;
263 bool first_pos_matched = false, first_ele_matched = false, first_v0_seen = false;
264 for (Vertex* vtx : vtxs) {
265 int i_ele{-1}, i_pos{-1};
266 for (int ipart = 0; ipart < vtx->getParticles().GetEntries(); ++ipart) {
267 int pdg_id = ((Particle*)vtx->getParticles().At(ipart))->getPDG();
268 if (pdg_id == 11)
269 i_ele = ipart;
270 else if (pdg_id == -11)
271 i_pos = ipart;
272 }
273 if (i_ele < 0 || i_pos < 0) continue;
274 n_v0++;
275
276 Particle* e = dynamic_cast<Particle*>(vtx->getParticles().At(i_ele));
277 Particle* p = dynamic_cast<Particle*>(vtx->getParticles().At(i_pos));
278 auto eEcal = e->getTrack().getPositionAtEcal();
279 auto pEcal = p->getTrack().getPositionAtEcal();
280 double pdr, pE, pt, edr, eE, et;
281 nearest_energetic(pEcal, pdr, pE, pt);
282 nearest_energetic(eEcal, edr, eE, et);
283 bool pmatch = (pdr < drMatch_);
284 bool ematch = (edr < drMatch_);
285 any_pos_matched = any_pos_matched || pmatch;
286 any_ele_matched = any_ele_matched || ematch;
287 if (not first_v0_seen) {
288 first_v0_seen = true;
289 first_pos_matched = pmatch;
290 first_ele_matched = ematch;
291 }
292 if (pdr < best_pos_dr) { // best v0 = positron closest to an energetic cluster
293 best_pos_dr = pdr;
294 chosen = vtx;
295 chosen_i_ele = i_ele;
296 chosen_i_pos = i_pos;
297 }
298 }
299 has_v0 = (chosen != nullptr);
300
301 // ---- track-level cluster match (decouple tracking vs vertexing) ----
302 // Independent of any v0: does a standalone track (KalmanFullTracks) point at an energetic
303 // cluster? If a positron TRACK matches but no v0 positron does, the loss is VERTEXING; if
304 // no track matches either, the loss is TRACKING/acceptance.
305 bool trk_pos_matched = false, trk_ele_matched = false;
306 bool has_orphan_cluster = false;
307 double orphan_clu_x = -9999., orphan_clu_y = -9999., orphan_clu_E = -9999.;
308 if (bus_.has(trkColl_)) {
309 const auto& tracks{bus_.get<std::vector<Track*>>(trkColl_)};
310 for (Track* trk : tracks) {
311 auto tEcal = trk->getPositionAtEcal();
312 if (tEcal.size() < 2) continue;
313 double dr, E, t;
314 nearest_energetic(tEcal, dr, E, t);
315 if (dr < drMatch_) {
316 if (trk->getCharge() > 0) trk_pos_matched = true;
317 else if (trk->getCharge() < 0) trk_ele_matched = true;
318 }
319 }
320
321 // orphan energetic clusters: energetic cluster with NO track (any charge) within
322 // drMatch_. Keep the highest-E orphan -> the lost (positron) cluster for the x,y map.
323 for (const auto* cl : allClusters) {
324 if (cl->getEnergy() <= eventClusterThresh_) continue;
325 auto cp = cl->getPosition();
326 double min_trk_dr = 9999.;
327 for (Track* trk : tracks) {
328 auto tE = trk->getPositionAtEcal();
329 if (tE.size() < 2) continue;
330 double dr = std::sqrt((tE[0] - cp[0]) * (tE[0] - cp[0]) +
331 (tE[1] - cp[1]) * (tE[1] - cp[1]));
332 if (dr < min_trk_dr) min_trk_dr = dr;
333 }
334 if (min_trk_dr > drMatch_) { // no track points at this energetic cluster
335 has_orphan_cluster = true;
336 if (cl->getEnergy() > orphan_clu_E) {
337 orphan_clu_E = cl->getEnergy();
338 orphan_clu_x = cp[0];
339 orphan_clu_y = cp[1];
340 }
341 }
342 }
343 }
344 // "tracking/acceptance" scenario for the positron: an energetic cluster exists but no
345 // positron track points at any energetic cluster.
346 bool pos_acceptance_loss = event_has_cluster && not trk_pos_matched;
347
348 bus_.set("trk_pos_matched", trk_pos_matched);
349 bus_.set("trk_ele_matched", trk_ele_matched);
350 bus_.set("n_v0", n_v0);
351 bus_.set("pos_acceptance_loss", pos_acceptance_loss);
352 bus_.set("has_orphan_cluster", has_orphan_cluster);
353 bus_.set<double>("orphan_clu_x", orphan_clu_x);
354 bus_.set<double>("orphan_clu_y", orphan_clu_y);
355 bus_.set<double>("orphan_clu_E", orphan_clu_E);
356
357 if (has_v0) {
358 Particle ele = *dynamic_cast<Particle*>(chosen->getParticles().At(chosen_i_ele));
359 Particle pos = *dynamic_cast<Particle*>(chosen->getParticles().At(chosen_i_pos));
360 Track ele_trk = ele.getTrack();
361 Track pos_trk = pos.getTrack();
362
363 TVector3 ele_p(ele_trk.getMomentum()[0], ele_trk.getMomentum()[1], ele_trk.getMomentum()[2]);
364 TVector3 pos_p(pos_trk.getMomentum()[0], pos_trk.getMomentum()[1], pos_trk.getMomentum()[2]);
365 bus_.set("ele_track_p", ele_p);
366 bus_.set("pos_track_p", pos_p);
367
368 auto eleEcal = ele_trk.getPositionAtEcal();
369 auto posEcal = pos_trk.getPositionAtEcal();
370 bus_.set<double>("ele_ecal_x", eleEcal[0]);
371 bus_.set<double>("ele_ecal_y", eleEcal[1]);
372 bus_.set<double>("ele_ecal_z", eleEcal[2]);
373 bus_.set<double>("pos_ecal_x", posEcal[0]);
374 bus_.set<double>("pos_ecal_y", posEcal[1]);
375 bus_.set<double>("pos_ecal_z", posEcal[2]);
376
377 // fiducial: use AnaHelpers::IsECalFiducial on the track projection at ECal
378 auto fiducial_at = [&](const std::vector<double>& xyz) -> bool {
379 float p[3] = {(float)xyz[0], (float)xyz[1], (float)xyz[2]};
380 CalCluster c;
381 c.setPosition(p);
382 return _ah->IsECalFiducial(&c);
383 };
384 ele_fiducial = fiducial_at(eleEcal);
385 pos_fiducial = fiducial_at(posEcal);
386 both_tracks_fiducial = ele_fiducial && pos_fiducial;
387
388 // associated clusters: a missing cluster has the sentinel energy (-9999)
389 CalCluster ele_clu = ele.getCluster();
390 CalCluster pos_clu = pos.getCluster();
391 ele_has_cluster = ele_clu.getEnergy() > clusterEnergyThresh_;
392 pos_has_cluster = pos_clu.getEnergy() > clusterEnergyThresh_;
393 pos_matched_cluster = pos_clu.getEnergy() > eventClusterThresh_;
394 pos_clu_E_dbg = pos_clu.getEnergy();
395 if (ele_has_cluster) {
396 bus_.set<double>("ele_clu_E", ele_clu.getEnergy());
397 bus_.set<double>("ele_clu_x", ele_clu.getPosition()[0]);
398 bus_.set<double>("ele_clu_y", ele_clu.getPosition()[1]);
399 bus_.set<double>("ele_clu_time", ele_clu.getTime() - calTimeOffset_);
400 bus_.set("ele_clu_nhits", ele_clu.getNHits());
401 }
402 if (pos_has_cluster) {
403 bus_.set<double>("pos_clu_E", pos_clu.getEnergy());
404 bus_.set<double>("pos_clu_x", pos_clu.getPosition()[0]);
405 bus_.set<double>("pos_clu_y", pos_clu.getPosition()[1]);
406 bus_.set<double>("pos_clu_time", pos_clu.getTime() - calTimeOffset_);
407 bus_.set("pos_clu_nhits", pos_clu.getNHits());
408 }
409
410 // dr-match of the BEST v0's tracks to the nearest energetic cluster (for dbg/printout)
411 double pos_dr, pos_dr_E, pos_dr_t;
412 double ele_dr, ele_dr_E, ele_dr_t;
413 nearest_energetic(posEcal, pos_dr, pos_dr_E, pos_dr_t);
414 nearest_energetic(eleEcal, ele_dr, ele_dr_E, ele_dr_t);
415 bus_.set<double>("pos_nearest_clu_dr", pos_dr);
416 bus_.set<double>("pos_nearest_clu_E", pos_dr_E);
417 bus_.set<double>("pos_nearest_clu_time", pos_dr_t);
418 pos_dr_matched = (pos_dr < drMatch_);
419 ele_dr_matched = (ele_dr < drMatch_);
420 pos_nearest_dr_dbg = pos_dr;
421 pos_dr_E_dbg = pos_dr_E;
422 ele_nearest_dr_dbg = ele_dr;
423 ele_dr_E_dbg = ele_dr_E;
424 }
425
426 bus_.set("truth_epem_found", truth_epem_found);
427 bus_.set("has_v0", has_v0);
428 bus_.set("ele_fiducial", ele_fiducial);
429 bus_.set("pos_fiducial", pos_fiducial);
430 bus_.set("both_tracks_fiducial", both_tracks_fiducial);
431 bus_.set("ele_has_cluster", ele_has_cluster);
432 bus_.set("pos_has_cluster", pos_has_cluster);
433
434 event_cf_.apply("has_v0", has_v0);
435 event_cf_.fill_nm1("has_v0", has_v0 ? 1 : 0);
436 event_cf_.apply("both_tracks_fiducial", both_tracks_fiducial);
437 event_cf_.fill_nm1("both_tracks_fiducial", both_tracks_fiducial ? 1 : 0);
438 event_cf_.apply("pos_has_cluster", pos_has_cluster);
439 event_cf_.fill_nm1("pos_has_cluster", pos_has_cluster ? 1 : 0);
440
441 if (has_v0) n_has_v0_++;
442 if (n_v0 > 1) n_multi_v0_++;
443 if (first_pos_matched) n_first_pos_matched_++;
444 if (first_ele_matched) n_first_ele_matched_++;
445 if (any_pos_matched) n_any_pos_matched_++;
446 if (any_ele_matched) n_any_ele_matched_++;
447 if (trk_pos_matched) n_trk_pos_matched_++;
448 if (trk_ele_matched) n_trk_ele_matched_++;
449 if (trk_pos_matched && not any_pos_matched) n_trkpos_no_v0pos_++;
450
451 // per-event terminal dump (first 40 triggered events) so we can inspect by eye.
452 // Compares v0-level vs track-level cluster match to separate tracking from vertexing.
453 if (n_printed_ < 40) {

458 any_pos_matched ? 1 : 0, any_ele_matched ? 1 : 0,

462 }

465 output_tree_->Fill();
