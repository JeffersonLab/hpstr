33void ClusterEfficiency2021::initialize(TTree* tree) {
34 _ah = std::make_shared<AnaHelpers>();
35
36 bus_.board_input<EventHeader>(tree, "EventHeader");
37 bus_.board_input<TSData>(tree, "TSBank");
38 if (not trkColl_.empty()) bus_.board_input<std::vector<Track*>>(tree, trkColl_);
39 bus_.board_input<std::vector<Vertex*>>(tree, vtxColl_);

43 // efficiency chain (quick at-a-glance, in addition to the tree)
44 event_cf_.add("single_trigger", 2, -0.5, 1.5);

46 event_cf_.add("has_v0", 2, -0.5, 1.5);
47 event_cf_.add("both_tracks_fiducial", 2, -0.5, 1.5);
48 event_cf_.add("pos_has_cluster", 2, -0.5, 1.5);
49 event_cf_.init();
50 event_cf_.set_label_names({"readout", "single2 or 3 trigger", "truth e^{+}e^{-} found",
51 "N_{vtx} #geq 1", "both tracks fiducial", "e^{+} has cluster"});

102bool ClusterEfficiency2021::process(IEvent*) {
103 const auto& tsbank{bus_.get<TSData>("TSBank")};
104 const auto& eh{bus_.get<EventHeader>("EventHeader")};
105
106 event_cf_.begin_event();
107
108 // Re-apply trigger decision: data contains other triggers; MC is produced with only
109 // this trigger but the event header is not updated, so the bits are still meaningful.
110 bool single2 = tsbank.isSingle2Trigger() == 1;
111 bool single3 = tsbank.isSingle3Trigger() == 1;
112 bool single_trigger = single2 || single3;
113 event_cf_.apply("single_trigger", single_trigger);
114 event_cf_.fill_nm1("single_trigger", single_trigger ? 1 : 0);
115 if (not event_cf_.keep()) return true;
116
117 // ----- defaults (sentinels) -----
118 bus_.set("weight", 1.);
119 bus_.set("run", eh.getRunNumber());
120 bus_.set("event", eh.getEventNumber());
121 bus_.set("single2", single2);
122 bus_.set("single3", single3);
123
124 bus_.set("ele_truth_p", TVector3(0., 0., 0.));
125 bus_.set("pos_truth_p", TVector3(0., 0., 0.));
126 bus_.set("ele_truth_E", -9999.);
127 bus_.set("pos_truth_E", -9999.);
128 bus_.set("true_vertex_invM", -9999.);
129 bus_.set("true_vertex_psum", -9999.);
130 bus_.set("true_decay_len", -9999.);
131 bus_.set("true_betagamma", -9999.);
132
133 bus_.set("ele_track_p", TVector3(0., 0., 0.));
134 bus_.set("pos_track_p", TVector3(0., 0., 0.));
135 for (const auto& name : {"ele_ecal_x", "ele_ecal_y", "ele_ecal_z",
136 "pos_ecal_x", "pos_ecal_y", "pos_ecal_z",
137 "ele_clu_E", "ele_clu_x", "ele_clu_y", "ele_clu_time",
138 "pos_clu_E", "pos_clu_x", "pos_clu_y", "pos_clu_time",
139 "pos_nearest_clu_dr", "pos_nearest_clu_E", "pos_nearest_clu_time"}) {
140 bus_.set<double>(name, -9999.);
141 }
142 bus_.set("ele_clu_nhits", -1);
143 bus_.set("pos_clu_nhits", -1);
144
145 bool truth_epem_found = false, has_v0 = false;
146 bool ele_fiducial = false, pos_fiducial = false, both_tracks_fiducial = false;
147 bool ele_has_cluster = false, pos_has_cluster = false;
148
149 // ===================== truth block =====================
150 const std::vector<MCParticle*>* mcParticles = nullptr;
151 if (bus_.has(mcColl_)) {
152 mcParticles = &bus_.get<std::vector<MCParticle*>>(mcColl_);
153 MCParticle* mother{nullptr};
154 ROOT::Math::PxPyPzEVector trueEleP, truePosP;
155 bool foundEle = false, foundPos = false;
156 for (MCParticle* ptr : *mcParticles) {
157 std::vector<double> lP = ptr->getMomentum();
158 if (ptr->getPDG() == apPDG_ || (isSimpSignal_ && ptr->getPDG() == 625)) {
159 mother = ptr;
160 } else if (ptr->getPDG() == 11 && ptr->getMomPDG() == signalMomPDG_) {
161 trueEleP = ROOT::Math::PxPyPzEVector(lP.at(0), lP.at(1), lP.at(2), ptr->getEnergy());
162 foundEle = true;
163 } else if (ptr->getPDG() == -11 && ptr->getMomPDG() == signalMomPDG_) {
164 truePosP = ROOT::Math::PxPyPzEVector(lP.at(0), lP.at(1), lP.at(2), ptr->getEnergy());
165 foundPos = true;
166 }
167 }
168 truth_epem_found = foundEle && foundPos;
169
170 if (foundEle) {
171 bus_.set("ele_truth_p", TVector3(trueEleP.Px(), trueEleP.Py(), trueEleP.Pz()));
172 bus_.set("ele_truth_E", trueEleP.E());
173 }
174 if (foundPos) {
175 bus_.set("pos_truth_p", TVector3(truePosP.Px(), truePosP.Py(), truePosP.Pz()));
176 bus_.set("pos_truth_E", truePosP.E());
177 }
178 if (foundEle && foundPos) {
179 bus_.set("true_vertex_invM", (trueEleP + truePosP).M());
180 bus_.set("true_vertex_psum", trueEleP.P() + truePosP.P());
181 }
182
183 if (mother) {
184 std::vector<double> vtxPos = mother->getVertexPosition();
185 std::vector<double> endPos = mother->getEndPoint();
186 double dx = endPos[0] - vtxPos[0];
187 double dy = endPos[1] - vtxPos[1];
188 double dz = endPos[2] - vtxPos[2];
189 bus_.set("true_decay_len", std::sqrt(dx * dx + dy * dy + dz * dz));
190
191 std::vector<double> mMom = mother->getMomentum();
192 double mp = std::sqrt(mMom[0] * mMom[0] + mMom[1] * mMom[1] + mMom[2] * mMom[2]);
193 if (mother->getMass() > 0) bus_.set("true_betagamma", mp / mother->getMass());
194 }
195 }
196 event_cf_.apply("truth_epem_found", isData_ ? true : truth_epem_found);
197 event_cf_.fill_nm1("truth_epem_found", truth_epem_found ? 1 : 0);
198
199 // ===================== reco block =====================
200 const auto& vtxs{bus_.get<std::vector<Vertex*>>(vtxColl_)};
201 const auto& allClusters{bus_.get<std::vector<CalCluster*>>("RecoEcalClusters")};
202
203 // pick a v0: prefer one whose electron/positron tracks truth-match e-/e+, else the first one
204 Vertex* chosen{nullptr};
205 int chosen_i_ele{-1}, chosen_i_pos{-1};
206 for (Vertex* vtx : vtxs) {
207 int i_ele{-1}, i_pos{-1};
208 for (int ipart = 0; ipart < vtx->getParticles().GetEntries(); ++ipart) {
209 int pdg_id = ((Particle*)vtx->getParticles().At(ipart))->getPDG();
210 if (pdg_id == 11)
211 i_ele = ipart;
212 else if (pdg_id == -11)
213 i_pos = ipart;
214 }
215 if (i_ele < 0 || i_pos < 0) continue;
216 if (chosen == nullptr) {
217 chosen = vtx;
218 chosen_i_ele = i_ele;
219 chosen_i_pos = i_pos;
220 }
221 if (mcParticles) {
222 Particle* e = dynamic_cast<Particle*>(vtx->getParticles().At(i_ele));
223 Particle* p = dynamic_cast<Particle*>(vtx->getParticles().At(i_pos));
224 Track et = e->getTrack();
225 Track pt = p->getTrack();
226 if (utils::getTruthPDG(et, mcParticles) == 11 && utils::getTruthPDG(pt, mcParticles) == -11) {
227 chosen = vtx;
228 chosen_i_ele = i_ele;
229 chosen_i_pos = i_pos;
230 break;
231 }
232 }
233 }
234 has_v0 = (chosen != nullptr);
235
236 if (has_v0) {
237 Particle ele = *dynamic_cast<Particle*>(chosen->getParticles().At(chosen_i_ele));
238 Particle pos = *dynamic_cast<Particle*>(chosen->getParticles().At(chosen_i_pos));
239 Track ele_trk = ele.getTrack();
240 Track pos_trk = pos.getTrack();
241
242 TVector3 ele_p(ele_trk.getMomentum()[0], ele_trk.getMomentum()[1], ele_trk.getMomentum()[2]);
243 TVector3 pos_p(pos_trk.getMomentum()[0], pos_trk.getMomentum()[1], pos_trk.getMomentum()[2]);
244 bus_.set("ele_track_p", ele_p);
245 bus_.set("pos_track_p", pos_p);
246
247 auto eleEcal = ele_trk.getPositionAtEcal();
248 auto posEcal = pos_trk.getPositionAtEcal();
249 bus_.set<double>("ele_ecal_x", eleEcal[0]);
250 bus_.set<double>("ele_ecal_y", eleEcal[1]);
251 bus_.set<double>("ele_ecal_z", eleEcal[2]);
252 bus_.set<double>("pos_ecal_x", posEcal[0]);
253 bus_.set<double>("pos_ecal_y", posEcal[1]);
254 bus_.set<double>("pos_ecal_z", posEcal[2]);
255
256 // fiducial: use AnaHelpers::IsECalFiducial on the track projection at ECal
257 auto fiducial_at = [&](const std::vector<double>& xyz) -> bool {
258 float p[3] = {(float)xyz[0], (float)xyz[1], (float)xyz[2]};
259 CalCluster c;
260 c.setPosition(p);
261 return _ah->IsECalFiducial(&c);
262 };
263 ele_fiducial = fiducial_at(eleEcal);
264 pos_fiducial = fiducial_at(posEcal);
265 both_tracks_fiducial = ele_fiducial && pos_fiducial;
266
267 // associated clusters: a missing cluster has the sentinel energy (-9999)
268 CalCluster ele_clu = ele.getCluster();
269 CalCluster pos_clu = pos.getCluster();
270 ele_has_cluster = ele_clu.getEnergy() > clusterEnergyThresh_;
271 pos_has_cluster = pos_clu.getEnergy() > clusterEnergyThresh_;
272 if (ele_has_cluster) {
273 bus_.set<double>("ele_clu_E", ele_clu.getEnergy());
274 bus_.set<double>("ele_clu_x", ele_clu.getPosition()[0]);
275 bus_.set<double>("ele_clu_y", ele_clu.getPosition()[1]);
276 bus_.set<double>("ele_clu_time", ele_clu.getTime() - calTimeOffset_);
277 bus_.set("ele_clu_nhits", ele_clu.getNHits());
278 }
279 if (pos_has_cluster) {
280 bus_.set<double>("pos_clu_E", pos_clu.getEnergy());
281 bus_.set<double>("pos_clu_x", pos_clu.getPosition()[0]);
282 bus_.set<double>("pos_clu_y", pos_clu.getPosition()[1]);
283 bus_.set<double>("pos_clu_time", pos_clu.getTime() - calTimeOffset_);
284 bus_.set("pos_clu_nhits", pos_clu.getNHits());
285 }
286
287 // standalone-cluster cross-check: nearest RecoEcalCluster to the positron track
288 // projection at ECal. Reveals cases where a cluster exists but was not associated.
289 double min_dr = 9999.0, min_dr_E = -9999.0, min_dr_t = -9999.0;
290 for (const auto* cl : allClusters) {
291 auto clPos = cl->getPosition();
292 double dx = clPos[0] - posEcal[0];
293 double dy = clPos[1] - posEcal[1];
294 double dr = std::sqrt(dx * dx + dy * dy);
295 if (dr < min_dr) {
296 min_dr = dr;
297 min_dr_E = cl->getEnergy();
298 min_dr_t = cl->getTime() - calTimeOffset_;
299 }
300 }
301 if (not allClusters.empty()) {
302 bus_.set<double>("pos_nearest_clu_dr", min_dr);
303 bus_.set<double>("pos_nearest_clu_E", min_dr_E);
304 bus_.set<double>("pos_nearest_clu_time", min_dr_t);
305 }
306 }
307
308 bus_.set("truth_epem_found", truth_epem_found);
309 bus_.set("has_v0", has_v0);
310 bus_.set("ele_fiducial", ele_fiducial);
311 bus_.set("pos_fiducial", pos_fiducial);
312 bus_.set("both_tracks_fiducial", both_tracks_fiducial);
313 bus_.set("ele_has_cluster", ele_has_cluster);
314 bus_.set("pos_has_cluster", pos_has_cluster);
315
316 event_cf_.apply("has_v0", has_v0);
317 event_cf_.fill_nm1("has_v0", has_v0 ? 1 : 0);
318 event_cf_.apply("both_tracks_fiducial", both_tracks_fiducial);
319 event_cf_.fill_nm1("both_tracks_fiducial", both_tracks_fiducial ? 1 : 0);
320 event_cf_.apply("pos_has_cluster", pos_has_cluster);
321 event_cf_.fill_nm1("pos_has_cluster", pos_has_cluster ? 1 : 0);
322
323 // one row per triggered signal event (sentinels where reco objects are absent)
324 output_tree_->Fill();
325 return true;
326}
