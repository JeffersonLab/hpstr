19void PreselectAndCategorizeSimp2016::configure(const ParameterSet& parameters) {
20 auto pSmearingFile = parameters.getString("pSmearingFile");
21
22 if (not pSmearingFile.empty()) {
23 // just using the same seed=42 for now
24 std::cout << "Loading momentum smearing from " << pSmearingFile << std::endl;
25 smearingTool_ = std::make_shared<TrackSmearingTool>(pSmearingFile,true);
26 }
27
28 auto beamPosCfg = parameters.getString("beamPosCfg");
29 if (not beamPosCfg.empty()) {
30 std::cout << "Loading beamspot corrections from " << beamPosCfg << std::endl;
31 auto beamspot_corrections_json = preselect_json_load(beamPosCfg);
32 for (const auto& [run, entry]: beamspot_corrections_json.items()) {
33 beamspot_corrections_[std::stoi(run)] = {
34 entry["beamspot_x"],
35 entry["beamspot_y"],
36 entry["beamspot_z"]
37 };
49 }
50 }
51
52 auto v0ProjectionFitsCfg = parameters.getString("v0ProjectionFitsCfg");
53 if (not v0ProjectionFitsCfg.empty()) {
54 std::cout << "Loading projection fits from " << v0ProjectionFitsCfg << std::endl;
55 v0proj_fits_ = preselect_json_load(v0ProjectionFitsCfg);
56 }
57
58 auto trackBiasCfg = parameters.getString("trackBiasCfg");
59 if (not trackBiasCfg.empty()) {
60 std::cout << "Loading track corrections from " << trackBiasCfg << std::endl;
61 auto track_corr = preselect_json_load(trackBiasCfg);
62 std::cout << track_corr << std::endl;
63
64 for (const auto& [name, corr]: track_corr.items()) {
65 // validate names of corrections, Track::applyCorrection silently ignores
66 // names that don't exactly match the names within it, so we check for
67 // misspellings once here
68 if (name != "track_z0" and name != "track_time") {
69 std::cerr << "WARNING: Unrecognized track correction "
70 << name << " : " << corr
71 << " will be skipped."
72 << std::endl;
73 continue;
74 }
75 track_corrections_[name] = corr;
76 }
77 }
78
79 calTimeOffset_ = parameters.getDouble("calTimeOffset");
80 isData_ = parameters.getInteger("isData") != 0;
81 isSignal_ = parameters.getInteger("isSignal") != 0;
82}

157bool PreselectAndCategorizeSimp2016::process(IEvent*) {
158 const auto& eh{bus_.get<EventHeader>("EventHeader")};
159 int run_number = eh.getRunNumber();
160
161 event_cf_.begin_event();
162
163 // re-apply trigger desicion to data sample since it contains
164 // other triggers within it
165 // MC is created with only this trigger AND the event header
166 // is not updated so we need to skip this check for MC
167 event_cf_.apply("pair1trigger", (not isData_ or eh.isPair1Trigger()));
168 if (not event_cf_.keep()) {
169 // we leave BEFORE filling the vertex counting histogram
170 // so that the vertex count histogram is relative to this trigger
171 // and does not include other triggers
172 return true;
173 }
174
175 const auto& vtxs{bus_.get<std::vector<Vertex*>>(vtxColl_)};
189 std::vector<std::tuple<Vertex,Particle,Particle>> preselected_vtx;
190 for (Vertex* vtx : vtxs) {
191 // access the indiviual Vertex, electron, and positron
192 // and add corrections to them before applying pre-selection
193 int i_ele{-1}, i_pos{-1};
194 for (int ipart = 0; ipart < vtx->getParticles().GetEntries(); ++ipart) {
195 int pdg_id = ((Particle*)vtx->getParticles().At(ipart))->getPDG();
196 if (pdg_id == 11) {
197 i_ele = ipart;
198 } else if (pdg_id == -11) {
199 i_pos = ipart;
200 }
201 }
202 if (i_ele < 0 or i_pos < 0) {
203 throw std::runtime_error("Vertex formed without either an electron or positron!");
204 }
205
206 Particle ele = *dynamic_cast<Particle*>(vtx->getParticles().At(i_ele));
207 Particle pos = *dynamic_cast<Particle*>(vtx->getParticles().At(i_pos));
208
209 // for both particles, get cluster, update, re-set cluster
210 CalCluster ele_clu = ele.getCluster();
211 CalCluster pos_clu = pos.getCluster();
212
213 ele_clu.setTime(ele_clu.getTime() - calTimeOffset_);
214 pos_clu.setTime(pos_clu.getTime() - calTimeOffset_);
215
216 ele.setCluster(&ele_clu);
217 pos.setCluster(&pos_clu);
218
219 Track ele_trk = ele.getTrack();
220 Track pos_trk = pos.getTrack();
221
222 // apply track_z0 and track_time corrections loaded from JSON
223 for (const auto& [name, corr]: track_corrections_) {
224 ele_trk.applyCorrection(name, corr);
225 pos_trk.applyCorrection(name, corr);
226 }
227
228 if (not beamspot_corrections_.empty()) {
245 auto bsit = beamspot_corrections_.find(eh.getRunNumber());
246 if (bsit == beamspot_corrections_.end()) {
247 // the current run number is not in the provided set of beamspots
248 // I don't think this should be the case, so I'm throwing an exception.
249 // We could also just pick the closest run number which is what
250 // the V0 projection fit code does.
251 throw std::runtime_error("The run number "+std::to_string(eh.getRunNumber())
252 +" is not in the loaded set of beamspot corrections.");
253 }
254 // assume iterator bsit is pointing to an element of beamspot_corrections_
255 // for the current run number.
256 // correction is the negative of the beamspot position to remove its shift
257 double bs_y = -1*bsit->second.at(1);
258 ele_trk.applyCorrection("track_z0", bs_y);
259 pos_trk.applyCorrection("track_z0", bs_y);
260 }
261
262 // smear track momentum
263 if (smearingTool_) {
264 double ele_smear = smearingTool_->updateWithSmearP(ele_trk);
265 double pos_smear = smearingTool_->updateWithSmearP(pos_trk);
266 smearingTool_->updateVertexWithSmearP(vtx, ele_smear, pos_smear);
267 }
268
269 // put tracks back into their particles
270 // with their new data
271 ele.setTrack(&ele_trk);
272 pos.setTrack(&pos_trk);
273
274 // corrections have been applied, now onto pre-selecting
275 double cluster_tdiff{abs(ele.getCluster().getTime() - pos.getCluster().getTime())};
276 int ele_nhits = ele.getTrack().getTrackerHitCount();
277 if (not ele.getTrack().isKalmanTrack()) ele_nhits*=2;
278 int pos_nhits = pos.getTrack().getTrackerHitCount();
279 if (not pos.getTrack().isKalmanTrack()) pos_nhits*=2;
280
281 double ele_track_cluster_tdiff{
282 abs(ele.getTrack().getTrackTime()-ele.getCluster().getTime())
283 };
284 double pos_track_cluster_tdiff{
285 abs(pos.getTrack().getTrackTime()-pos.getCluster().getTime())
286 };
287
288 TVector3 ele_mom(
289 ele.getTrack().getMomentum()[0],
290 ele.getTrack().getMomentum()[1],
291 ele.getTrack().getMomentum()[2]
292 );
293 TVector3 pos_mom(
294 pos.getTrack().getMomentum()[0],
295 pos.getTrack().getMomentum()[1],
296 pos.getTrack().getMomentum()[2]
297 );
298
299 vertex_cf_.begin_event();
300 vertex_cf_.apply("abs_ele_track_before_6ns", abs(ele.getTrack().getTrackTime()) <= 6.0);
301 vertex_cf_.apply("abs_pos_track_before_6ns", abs(pos.getTrack().getTrackTime()) <= 6.0);
302 vertex_cf_.apply("clusters_within_1.45ns", cluster_tdiff <= 1.45);
303 vertex_cf_.apply("ele_track_cluster_within_4ns", ele_track_cluster_tdiff <= 4.0);
304 vertex_cf_.apply("pos_track_cluster_within_4ns", pos_track_cluster_tdiff <= 4.0);
305 vertex_cf_.apply("ele_track_chi2ndf", ele.getTrack().getChi2Ndf() <= 20.0);
306 vertex_cf_.apply("pos_track_chi2ndf", pos.getTrack().getChi2Ndf() <= 20.0);
307 vertex_cf_.apply("electron_below_1.75GeV", ele.getTrack().getP() <= 1.75);
308 vertex_cf_.apply("ele_min_8_hits", ele_nhits >= 7);
309 vertex_cf_.apply("pos_min_8_hits", pos_nhits >= 7);
310 vertex_cf_.apply("vertex_chi2", vtx->getChi2() <= 20.0);
311 double vtxmaxp = (ele_mom+pos_mom).Mag();
312 vertex_cf_.apply("vtx_max_p_2.4GeV", vtxmaxp <= 2.4);
313
314 vertex_cf_.fill_nm1("abs_ele_track_before_6ns", abs(ele.getTrack().getTrackTime()));
315 vertex_cf_.fill_nm1("abs_pos_track_before_6ns", abs(pos.getTrack().getTrackTime()));
316 vertex_cf_.fill_nm1("clusters_within_1.45ns", cluster_tdiff);
317 vertex_cf_.fill_nm1("ele_track_cluster_within_4ns", ele_track_cluster_tdiff);
318 vertex_cf_.fill_nm1("pos_track_cluster_within_4ns", pos_track_cluster_tdiff);
319 vertex_cf_.fill_nm1("ele_track_chi2ndf", ele.getTrack().getChi2Ndf());
320 vertex_cf_.fill_nm1("pos_track_chi2ndf", pos.getTrack().getChi2Ndf());
321 vertex_cf_.fill_nm1("electron_below_1.75GeV", ele.getTrack().getP());
322 vertex_cf_.fill_nm1("ele_min_8_hits", ele_nhits);
323 vertex_cf_.fill_nm1("pos_min_8_hits", pos_nhits);
324 vertex_cf_.fill_nm1("vertex_chi2", vtx->getChi2());
325 vertex_cf_.fill_nm1("vtx_max_p_2.4GeV", vtxmaxp);
326
327 if (vertex_cf_.keep()) {
328 preselected_vtx.emplace_back(*vtx, ele, pos);
329 }
330 }
331
332 n_vertices_h_->Fill(vtxs.size(), preselected_vtx.size());
333 event_cf_.apply("at_least_one_vertex", preselected_vtx.size() > 0);
334 event_cf_.apply("no_extra_vertices", preselected_vtx.size() < 2);
335 if (not event_cf_.keep()) {
336 return true;
337 }
338
339 // correct number of vertices (i.e. only one)
340 // unpack the vector of vertices into the single elements
341 auto [ vtx, ele, pos ] = preselected_vtx.at(0);
342
343 // earliest layer hit categories
344 bool eleL1{false}, eleL2{false}, posL1{false}, posL2{false};
345 Track ele_trk{ele.getTrack()}, pos_trk{pos.getTrack()};
346 _ah->InnermostLayerCheck(&ele_trk, eleL1, eleL2);
347 _ah->InnermostLayerCheck(&pos_trk, posL1, posL2);
348 bus_.set("eleL1", eleL1);
349 bus_.set("eleL2", eleL2);
350 bus_.set("posL1", posL1);
351 bus_.set("posL2", posL2);
352
353 TVector3 ele_mom(
354 ele_trk.getMomentum()[0],
355 ele_trk.getMomentum()[1],
356 ele_trk.getMomentum()[2]
357 );
358 TVector3 pos_mom(
359 pos_trk.getMomentum()[0],
360 pos_trk.getMomentum()[1],
361 pos_trk.getMomentum()[2]
362 );
363 bus_.set("psum", ele_mom.Mag() + pos_mom.Mag());
364
365 // calculate target projection and its significance
366 if (not v0proj_fits_.empty()) {
367 double vtx_proj_x{-1.0}, vtx_proj_y{-1.0};
368 double vtx_proj_x_sig{-1.0}, vtx_proj_y_sig{-1.0};
369 double vtx_proj_sig{-1.0};
370
371 vtx_proj_sig = utils::v0_projection_to_target_significance(
372 v0proj_fits_, eh.getRunNumber(),
373 vtx_proj_x, vtx_proj_y, vtx_proj_x_sig, vtx_proj_y_sig,
374 vtx.getX(), vtx.getY(), vtx.getZ(),
375 vtx.getP().X(), vtx.getP().Y(), vtx.getP().Z()
376 );
377
378 bus_.set("vtx_proj_sig", vtx_proj_sig);
379 bus_.set("vtx_proj_x", vtx_proj_x);
380 bus_.set("vtx_proj_x_sig", vtx_proj_x_sig);
381 bus_.set("vtx_proj_y", vtx_proj_y);
382 bus_.set("vtx_proj_y_sig", vtx_proj_y_sig);
383 }
384
385 bus_.set("weight", 1.);
386 bus_.set("vertex", vtx);
387 bus_.set("ele", ele);
388 bus_.set("pos", pos);
389
399 if (bus_.has(mcColl_) and isSignal_) {
405 std::map<int, int> count_per_particle_id;
406 for (const auto& [layer_id, particle_id] : ele_trk.getMcpHits()) {
407 if (count_per_particle_id.find(particle_id) == count_per_particle_id.end()) {
408 count_per_particle_id[particle_id] = 0;
409 }
410 count_per_particle_id[particle_id]++;
411 }
412 int truth_ele_id{-1}, max_nhits{0};
413 for (const auto& [particle_id, count] : count_per_particle_id) {
414 if (count > max_nhits) {
415 truth_ele_id = particle_id;
416 max_nhits = count;
417 }
418 }
429 const auto& mc_ptr{bus_.get<std::vector<MCParticle*>>(mcColl_)};
430 MCParticle* vd{nullptr};
431 int n_vd{0};
432 bool ele_is_rad_ele{false};
433 for (MCParticle* ptr : mc_ptr) {
434 if (ptr->getPDG() == 625) {
435 n_vd++;
436 vd = ptr;
437 } else if (ptr->getID() == truth_ele_id) {
438 ele_is_rad_ele = (ptr->getMomPDG() == 625);
439 }
440 }
441 event_cf_.apply("at_least_one_true_vd", n_vd > 0);
442 event_cf_.apply("no_extra_true_vd", n_vd < 2);
443 if (not event_cf_.keep()) {
444 return true;
445 }
446 if (not vd) {
447 throw std::runtime_error(
448 "ERROR: Logic error: checked for VD earlier but there isn't one."
449 );
450 }
451 bus_.set("true_vd", *vd);
452 bus_.set("isRadEle", ele_is_rad_ele);
453 }
454 output_tree_->Fill();
455 return true;
456}
